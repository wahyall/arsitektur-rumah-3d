   342→function clamp01(v: number): number {
   343→  return v < 0 ? 0 : v > 1 ? 1 : v;

   346→interface Interval {
   347→  lo: number;
   348→  hi: number;
   349→}
   350→
   351→function subtractIntervals(base: Interval, cuts: Interval[]): Interval[] {
   352→  let pieces: Interval[] = [{ ...base }];
   353→  for (const c of cuts) {
   354→    const next: Interval[] = [];
   355→    for (const p of pieces) {
   356→      if (c.hi <= p.lo || c.lo >= p.hi) {
   357→        next.push(p);
   358→        continue;
   359→      }
   360→      if (c.lo > p.lo) next.push({ lo: p.lo, hi: c.lo });
   361→      if (c.hi < p.hi) next.push({ lo: c.hi, hi: p.hi });
   362→    }
   363→    pieces = next;
   364→  }
   365→  return pieces.filter((p) => p.hi - p.lo > 0.015); // buang irisan sangat tipis
   366→}

   368→/**
   369→ * Selesaikan sentinel lubang angin menjadi elevasi nyata.
   370→ * "atas"  -> sill = tinggi dinding - 60 cm (dekat puncak dinding, di bawah ring balk)
   371→ * "pelana"-> di segitiga pelana, menjorok sedikit di bawah puncak apex
   372→ */
   373→function resolveVent(
   374→  o: OpeningSpec,
   375→  t: { rbTop: number; apexAdd: number; tinggi: number },
   376→): OpeningSpec | null {
   377→  if (o.kind !== "vent") return o;
   378→  const hgt = Math.min(0.22, o.to - o.from); // tinggi roster (maks 22 cm)
   379→  if (o.sill === -1) {
   380→    const cap = t.rbTop + t.apexAdd;
   381→    const top = cap - 0.16;
   382→    const sill = top - hgt;
   383→    return sill < t.rbTop + 0.12 ? null : { ...o, sill, top }; // pelana terlalu landai
   384→  }
   385→  const sill = t.tinggi - 0.6;
   386→  return { ...o, sill, top: sill + hgt };
   387→}

   450→export function buildModel(config: HouseConfig): HouseModel {
   451→  const preset = LAYOUT_PRESETS.find((p) => p.id === config.preset) ?? LAYOUT_PRESETS[0];
   452→  const tL = config.tLuar / 1000;
   453→  const tD = config.tDalam / 1000;
   454→  const dims = BRICK[config.jenis];
   455→  const j = config.natMm / 1000;
   456→
   457→  // geometri atap pelana: ridge memanjang arah Y denah (dinding ujung pelana = depan/belakang)
   458→  const rad = (config.atapDeg * Math.PI) / 180;
   459→  const apexAdd = 3 * Math.tan(rad); // setengah bentang 3 m -> naik ridge
   460→  const pitch = Math.tan(rad);
   461→  const rbTop = config.tinggi + RB_H; // puncak ring balk = dasar dinding pelana
   462→
   463→  // ----- dinding runtime
   464→  const walls: WallRT[] = preset.walls.map((spec) => {
   465→    const t = spec.outer ? tL : tD;
   466→    let c: number;
   467→    if (spec.edge !== undefined) {
   468→      c = spec.edge === 0 ? t / 2 : 6 - t / 2;
   469→    } else {
   470→      c = spec.pos ?? 0;
   471→    }
   472→    const lo = typeof spec.from === "function" ? spec.from({ tL, tD }) : spec.from;
   473→    const hi = typeof spec.to === "function" ? spec.to({ tL, tD }) : spec.to;
   474→    const explodeDir: [number, number] =
   475→      spec.axis === "h"
   476→        ? spec.edge === 6 || (!spec.outer && (spec.pos ?? 0) < 3)
   477→          ? [0, 1]
   478→          : [0, -1]
   479→        : spec.edge === 6 || (!spec.outer && (spec.pos ?? 0) > 3)
   480→          ? [1, 0]
   481→          : [-1, 0];
   482→    return {
   483→      spec,
   484→      id: spec.id,
   485→      name: spec.name,
   486→      axis: spec.axis,
   487→      c,
   488→      lo,
   489→      hi,
   490→      t,
   491→      h: config.tinggi,
   492→      gable: spec.gable ? { ...spec.gable, apexAdd, baseZ: rbTop } : undefined,
   493→      openings: spec.openings
   494→        .map((o) => resolveVent(o, { rbTop, apexAdd, tinggi: config.tinggi }))
   495→        .filter((o): o is OpeningSpec => !!o)
   496→        .filter((o) => o.to > lo && o.from < hi && o.top > 0.2),
   497→      outer: spec.outer,
   498→      explodeDir,
   499→    };
   500→  });
   501→
   502→  // ----- susunan bata per dinding
   503→  const bricks: BrickInstance[] = [];
   504→  let outerPieces = 0;
   505→  let innerPieces = 0;
   506→  let wallFace = 0;
   507→  let bandVol = 0;
   508→  let brickVol = 0;
   509→  let wallFootprint = 0;
   510→  let wallLenTotal = 0;
   511→
   512→  const cellL = dims.L + j;
   513→  const cellH = dims.H + j;
   514→
   515→  for (const w of walls) {
   516→    const len = w.hi - w.lo;
   517→    wallLenTotal += len;
   518→    wallFootprint += len * w.t;
   519→
   520→    const hAt = w.gable
   521→      ? (x: number) => w.gable!.baseZ + w.gable!.apexAdd * clamp01(1 - Math.abs(x - w.gable!.apexAt) / w.gable!.halfSpan)
   522→      : () => w.h;
   523→
   524→    // luas muka satu sisi (net bukaan: tinggi bukaan = top - sill, dipotong tinggi
   525→    // dinding datar / pelana; band ring balk tidak dihitung karena beton)
   526→    const gableArea = w.gable ? w.gable.apexAdd * (w.hi - w.lo) * 0.5 : 0;
   527→    const hAtMid = (o: OpeningSpec) =>
   528→      Math.max(w.h, hAt((o.from + o.to) / 2));
   529→    const openingArea = w.openings.reduce(
   530→      (s, o) =>
   531→        s +
   532→        Math.max(0, Math.min(o.to, w.hi) - Math.max(o.from, w.lo)) *
   533→          Math.max(0, Math.min(o.top, hAtMid(o)) - o.sill),
   534→      0,
   535→    );
   536→    wallFace += w.h * len + gableArea - openingArea;
   537→    bandVol += (w.h * len + gableArea) * w.t - openingArea * w.t;
   538→
   539→    const topZ = w.gable ? w.gable.baseZ + w.gable.apexAdd : w.h;
   540→    const nCourses = Math.ceil(topZ / cellH);
   541→    for (let i = 0; i < nCourses; i++) {
   542→      const zc = i * cellH;
   543→      if (zc >= topZ - 0.001) break;
   544→
   545→      // interval yang boleh dipasangi bata pada lajur ini:
   546→      // 1) band ring balk (w.h .. baseZ) tidak dipasangi bata; di atas baseZ
   547→      //    bata mengikuti penyempitan segitiga pelana
   548→      if (w.gable && zc >= w.h && zc < w.gable.baseZ) continue; // zona ring balk
   549→
   550→      const allowed: Interval = { lo: w.lo, hi: w.hi };
   551→      if (w.gable && zc >= w.gable.baseZ) {
   552→        const d = w.gable.halfSpan * (1 - (zc - w.gable.baseZ) / w.gable.apexAdd);
   553→        if (d <= 0) continue;
   554→        allowed.lo = Math.max(w.lo, w.gable.apexAt - d);
   555→        allowed.hi = Math.min(w.hi, w.gable.apexAt + d);
   556→      }
   557→      if (allowed.hi - allowed.lo <= 0.015) continue;
   558→
   559→      // 2) bukaan yang memotong lajur ini secara vertikal
   560→      const activeCuts: Interval[] = w.openings
   561→        .filter((o) => o.sill < zc + dims.H - 0.005 && o.top > zc + 0.005)
   562→        .map((o) => ({ lo: Math.max(o.from, w.lo), hi: Math.min(o.to, w.hi) }));
   563→
   564→      // 3) isi bata dengan ikatan berselang (stretcher bond)
   565→      const phase = (i % 2) * cellL * 0.5;
   566→      let p = allowed.lo - phase;
   567→      while (p + dims.L <= allowed.lo) p += cellL;
   568→      for (; p < allowed.hi; p += cellL) {
   569→        const segs = subtractIntervals(
   570→          { lo: Math.max(p, allowed.lo), hi: Math.min(p + dims.L, allowed.hi) },
   571→          activeCuts,
   572→        );
   573→        for (const s of segs) {
   574→          const cx = (s.lo + s.hi) / 2;
   575→          // di bawah ring balk dipotong rata di tinggi dinding; di atas mengikuti kemiringan pelana
   576→          const hgt =
   577→            zc < w.h
   578→              ? Math.min(dims.H, Math.max(0, w.h - zc))
   579→              : Math.min(dims.H, Math.max(0, hAt(cx) - zc));
   580→          if (hgt <= 0.02) continue;
   581→          const frac = ((s.hi - s.lo) / dims.L) * (hgt / dims.H);
   582→          if (w.outer) outerPieces += frac;
   583→          else innerPieces += frac;
   584→          brickVol += frac * dims.L * dims.H * w.t;
   585→          bricks.push(
   586→            w.axis === "h"
   587→              ? { wallId: w.id, outer: w.outer, x: cx, y: w.c, z: zc, len: s.hi - s.lo, hgt, rotY: 0 }
   588→              : { wallId: w.id, outer: w.outer, x: w.c, y: cx, z: zc, len: s.hi - s.lo, hgt, rotY: Math.PI / 2 },
   589→          );
   590→        }
   591→      }
   592→    }
   593→  }
   594→
   595→  const mortarVol = Math.max(0, bandVol - brickVol);
   596→
   597→  // ----- luas bersih ruangan (setelah dikurangi tebal dinding)
   598→  interface Band {
   599→    axis: Axis;
   600→    x0: number;
   601→    x1: number;
   602→    y0: number;
   603→    y1: number;
   604→  }
   605→  const bands: Band[] = walls.map((w) =>
   606→    w.axis === "h"
   607→      ? { axis: w.axis, x0: w.lo, x1: w.hi, y0: w.c - w.t / 2, y1: w.c + w.t / 2 }
   608→      : { axis: w.axis, x0: w.c - w.t / 2, x1: w.c + w.t / 2, y0: w.lo, y1: w.hi },
   609→  );
   610→  const rooms: RoomNet[] = preset.rooms.map((r) => {
   611→    let xl = 0, xr = 0, yb = 0, yt = 0;
   612→    for (const b of bands) {
   613→      const yOverlap = Math.min(b.y1, r.y1) - Math.max(b.y0, r.y0);
   614→      const xOverlap = Math.min(b.x1, r.x1) - Math.max(b.x0, r.x0);
   615→      if (b.axis === "v") {
   616→        if (b.x0 <= r.x0 + 0.02 && b.x1 > r.x0 && yOverlap > 0.05) xl = Math.max(xl, b.x1 - r.x0);
   617→        if (b.x1 >= r.x1 - 0.02 && b.x0 < r.x1 && yOverlap > 0.05) xr = Math.max(xr, r.x1 - b.x0);
   618→      } else {
   619→        if (b.y0 <= r.y0 + 0.02 && b.y1 > r.y0 && xOverlap > 0.05) yb = Math.max(yb, b.y1 - r.y0);
   620→        if (b.y1 >= r.y1 - 0.02 && b.y0 < r.y1 && xOverlap > 0.05) yt = Math.max(yt, r.y1 - b.y0);
   621→      }
   622→    }
   623→    const wNet = Math.max(0, r.x1 - r.x0 - xl - xr);
   624→    const dNet = Math.max(0, r.y1 - r.y0 - yb - yt);
   625→    return { id: r.id, name: r.name, w: wNet, d: dNet, area: wNet * dNet, cx: (r.x0 + r.x1) / 2, cy: (r.y0 + r.y1) / 2 };
   626→  });
   627→
   628→  const netFloor = rooms.reduce((s, r) => s + r.area, 0);
   629→  const gross = 36;
   630→
   631→  // ----- kolom praktis 15/15: pada tiap pertemuan centerline dinding h x v
   632→  const seen = new Set<string>();
   633→  const koloms: KolomRT[] = [];
   634→  const tol = 0.15;
   635→  for (const h of walls) {
   636→    if (h.axis !== "h") continue;
   637→    for (const v of walls) {
   638→      if (v.axis !== "v") continue;
   639→      if (v.c >= h.lo - tol && v.c <= h.hi + tol && h.c >= v.lo - tol && h.c <= v.hi + tol) {
   640→        const key = `${v.c.toFixed(2)}|${h.c.toFixed(2)}`;
   641→        if (!seen.has(key)) {
   642→          seen.add(key);
   643→          koloms.push({ x: v.c, y: h.c });
   644→        }
   645→      }
   646→    }
   647→  }
   648→
   649→  // ----- material
   650→  const waste = config.wastePct / 100;
   651→  const mortarWithWaste = mortarVol * (1 + waste);
   652→  const semenZak = config.jenis === "merah" ? mortarWithWaste * 6.3 : 0;
   653→  const pasirM3 = config.jenis === "merah" ? mortarWithWaste * 1.0 : 0;
   654→  const perekatKg = config.jenis === "hebel" ? wallFace * 4 : 0;
   655→  const perekatZak = config.jenis === "hebel" ? perekatKg / 40 : 0;
   656→
   657→  // ----- atap pelana (kuda-kuda C75 jarak maks 1,2 m, genteng beton 10 pcs/m2)
   658→  const ov = config.overhang;
   659→  const yRidge = rbTop + apexAdd;
   660→  const slopeHalf = Math.hypot(3 + ov, apexAdd + ov * pitch);
   661→  const roofArea = 2 * slopeHalf * (6 + 2 * ov);
   662→  // jumlah kuda-kuda: seluruh bentang 6 m dengan jarak antar rangka maks 1,2 m
   663→  const kudaKuda = Math.max(4, Math.ceil(6 / KUDA_SPACING) + 1);
   664→  const rengPerSlope = Math.ceil(3 / 0.4) + 1;
   665→  const gentengPcs = Math.ceil(roofArea * 10);
   666→
   667→  const doors = walls.reduce((s, w) => s + w.openings.filter((o) => o.kind === "pintu").length, 0);
   668→  const windows = walls.reduce((s, w) => s + w.openings.filter((o) => o.kind === "jendela").length, 0);
   669→  const vents = walls.reduce((s, w) => s + w.openings.filter((o) => o.kind === "vent").length, 0);
   670→  const windowAreaM2 = walls.reduce(
   671→    (s, w) => s + w.openings.filter((o) => o.kind === "jendela").reduce((a, o) => a + (o.to - o.from) * (o.top - o.sill), 0),
   672→    0,
   673→  );
   674→
   675→  // ----- pondasi cakar ayam: buis di tiap titik kolom + grid tengah (jarak maks ~3 m)
   676→  let cakar: CakarAyamRT | null = null;
   677→  if (config.pondasi === "cakarayam") {
   678→    const pts: { x: number; y: number }[] = koloms.map((k) => ({ x: k.x, y: k.y }));
   679→    for (const gx of [1.5, 4.5]) for (const gy of [1.5, 4.5]) pts.push({ x: gx, y: gy });
   680→    const buisPts: { x: number; y: number }[] = [];
   681→    for (const p of pts) {
   682→      if (!buisPts.some((q) => Math.hypot(q.x - p.x, q.y - p.y) < 0.45)) buisPts.push(p);
   683→    }
   684→    const rOut = CAKAR.buisDia / 2;
   685→    const rIn = rOut - CAKAR.buisWall;
   686→    const buisM3 = buisPts.length * Math.PI * (rOut * rOut - rIn * rIn) * CAKAR.buisL;
   687→    const slabM3 = 36 * CAKAR.slabT;
   688→    cakar = {
   689→      slabM3,
   690→      buisCount: buisPts.length,
   691→      buisM3,
   692→      uruganM3: 36 * CAKAR.bed,
   693→      galianM3: 36 * CAKAR.bed + buisPts.length * Math.PI * rOut * rOut * CAKAR.buisL,
   694→      betonM3: slabM3 + buisM3,
   695→      buisPts,
   696→    };
   697→  }
   698→
   699→  // ----- kuantitas struktur (galian & pondasi mengikuti jenis pondasi)
   700→  const galianM3 =
   701→    config.pondasi === "cakarayam" && cakar
   702→      ? cakar.galianM3
   703→      : wallLenTotal * PONDASI_D * GALIAN_W; // galian 60 x 60 cm
   704→  const uruganPasirM3 = config.pondasi === "cakarayam" ? 0 : wallLenTotal * GALIAN_W * URUGAN_T;
   705→  const pondasiM3 = config.pondasi === "cakarayam" ? 0 : wallLenTotal * PONDASI_W * PONDASI_D;
   706→  const sloofM3 = wallLenTotal * SLOOF_W * RB_H;
   707→  const kolomM3 = koloms.length * KOLOM_S * KOLOM_S * (config.tinggi + RB_H);
   708→  const ringbalkM3 = wallLenTotal * SLOOF_W * RB_H;
   709→  const lisplankM = 2 * (6 + 2 * ov) + 2 * slopeHalf; // eaves + kedua ujung pelana
   710→  const talangM = 2 * (6 + 2 * ov);
   711→
   712→  return {
   713→    config,
   714→    walls,
   715→    brickDims: { L: dims.L, H: dims.H },
   716→    bricks,
   717→    count: {
   718→      outerPieces,
   719→      innerPieces,
   720→      totalPieces: outerPieces + innerPieces,
   721→      buyOuter: Math.ceil(outerPieces * (1 + waste)),
   722→      buyInner: Math.ceil(innerPieces * (1 + waste)),
   723→      totalBuy: Math.ceil(outerPieces * (1 + waste)) + Math.ceil(innerPieces * (1 + waste)),
   724→    },
   725→    mortar: {
   726→      volM3: mortarVol,
   727→      semenZak,
   728→      pasirM3,
   729→      perekatKg,
   730→      perekatZak,
   731→      natMm: config.natMm,
   732→    },
   733→    areas: {
   734→      gross,
   735→      netFloor,
   736→      rooms,
   737→      reduksiM2: gross - netFloor,
   738→      reduksiPct: ((gross - netFloor) / gross) * 100,
   739→      wallFace,
   740→      plasterArea: wallFace * 2,
   741→      wallFootprint,
   742→    },
   743→    openings: { doors, windows, vents, windowAreaM2 },
   744→    vent: {
   745→      bukaanM2: windowAreaM2,
   746→      pctLantai: (windowAreaM2 / Math.max(0.01, netFloor)) * 100,
   747→      okSni: windowAreaM2 >= 0.05 * netFloor,
   748→      okIdeal: windowAreaM2 >= 0.1 * netFloor,
   749→    },
   750→    roof: { areaM2: roofArea, apexAdd, overhang: ov, deg: config.atapDeg, pitch, yRidge, kudaKuda, rengPerSlope, gentengPcs },
   751→    wallLenTotal,
   752→    koloms,
   753→    structure: { galianM3, uruganPasirM3, pondasiM3, sloofM3, kolomM3, ringbalkM3, lisplankM, talangM, cakar },
   754→  };
   755→}

   845→export function buildRab(model: HouseModel, prices: PriceBook, upahPct: number): RabSummary {
   846→  const cfg = model.config;
   847→  const rows: RabRow[] = [];
   848→  const push = (id: string, divisi: string, uraian: string, spek: string, qty: number, unit: string, exact: boolean, priceKey?: string) => {
   849→    const key = priceKey ?? id;
   850→    const harga = prices[key] ?? 0;
   851→    rows.push({ id, divisi, uraian, spek, qty, unit, harga, subtotal: qty * harga, exact, priceKey: key });
   852→  };
   853→
   854→  push("bouwplank", "d1", "Pengukuran & bouwplank", "Profil kayu + patok as bangunan", 1, "lot", false);
   855→
   856→  if (cfg.pondasi === "cakarayam" && model.structure.cakar) {
   857→    const c = model.structure.cakar;
   858→    push("galian", "d2", "Galian titik buis + lantai kerja pelat", "Lubang buis Ø40 cm sedalam 80 cm + perataan 5 cm di seluruh pelat", Math.ceil(c.galianM3 * 10) / 10, "m3", true);
   859→    push("urugan", "d2", "Urugan pasir bawah pelat", "Pasir setempat 5 cm, dipadatkan sebelum cor pelat", Math.ceil(c.uruganM3 * 10) / 10, "m3", true, "urugan");
   860→    push("cakarBeton", "d2", "Pondasi cakar ayam (sistem Sedyatmo)", `Pelat 6x6 m tebal 10 cm wiremesh D8 + ${c.buisCount} buis beton Ø40 cm panjang 80 cm, cor monolit`, Math.ceil(c.betonM3 * 100) / 100, "m3", true, "cakarBeton");
   861→  } else {
   862→    push("galian", "d2", "Galian tanah pondasi", "Galian 60 x 60 cm sedalam 60 cm sepanjang jalur dinding", Math.ceil(model.structure.galianM3 * 10) / 10, "m3", true);
   863→    push("uruganPasir", "d2", "Urugan pasir bawah pondasi", "Pasir setempat 10 cm dipadatkan di dasar galian", Math.ceil(model.structure.uruganPasirM3 * 10) / 10, "m3", true, "urugan");
   864→    push("pondasi", "d2", "Pondasi batu kali + campuran 1:5", "Strip 40 x 60 cm (kedalaman 60 cm)", Math.ceil(model.structure.pondasiM3 * 10) / 10, "m3", true);
   865→  }
