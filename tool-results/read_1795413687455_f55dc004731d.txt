   542→function clamp01(v: number): number {
   543→  return v < 0 ? 0 : v > 1 ? 1 : v;
   544→}
   545→
   546→interface Interval {
   547→  lo: number;
   548→  hi: number;
   549→}
   550→
   551→function subtractIntervals(base: Interval, cuts: Interval[]): Interval[] {
   552→  let pieces: Interval[] = [{ ...base }];
   553→  for (const c of cuts) {
   554→    const next: Interval[] = [];
   555→    for (const p of pieces) {
   556→      if (c.hi <= p.lo || c.lo >= p.hi) {
   557→        next.push(p);
   558→        continue;
   559→      }
   560→      if (c.lo > p.lo) next.push({ lo: p.lo, hi: c.lo });
   561→      if (c.hi < p.hi) next.push({ lo: c.hi, hi: p.hi });
   562→    }
   563→    pieces = next;
   564→  }
   565→  return pieces.filter((p) => p.hi - p.lo > 0.015); // buang irisan sangat tipis
   566→}

   568→/**
   569→ * Selesaikan sentinel lubang angin menjadi elevasi nyata.
   570→ * "atas"  -> sill = tinggi dinding - 60 cm (dekat puncak dinding, di bawah ring balk)
   571→ * "pelana"-> di segitiga pelana, menjorok sedikit di bawah puncak apex
   572→ */
   573→function resolveVent(
   574→  o: OpeningSpec,
   575→  t: { rbTop: number; apexAdd: number; tinggi: number },
   576→): OpeningSpec | null {
   577→  if (o.kind !== "vent") return o;
   578→  const hgt = Math.min(0.22, o.to - o.from); // tinggi roster (maks 22 cm)
   579→  if (o.sill === -1) {
   580→    const cap = t.rbTop + t.apexAdd;
   581→    const top = cap - 0.16;
   582→    const sill = top - hgt;
   583→    return sill < t.rbTop + 0.12 ? null : { ...o, sill, top }; // pelana terlalu landai
   584→  }
   585→  const sill = t.tinggi - 0.6;
   586→  return { ...o, sill, top: sill + hgt };
   587→}

   613→  conduitM: number; // conduit PVC 20 mm
   614→  kabelLampuM: number; // circuit lampu (NYM 2x1,5)
   615→  kabelKontakM: number; // circuit stop kontak (NYM 2x2,5)
   616→  titikLampu: number;
   617→  saklar: number;
   618→  stopkontak: number;
   619→  kloset: number;
   620→  shower: number;
   621→  kran: number;
   622→  drain: number;
   623→  routes: MepRouteRT[];
   624→  points: MepPointRT[];
   625→  septic: { x: number; z: number };
   626→  resapan: { x: number; z: number };
   627→}
   628→
   629→function polylineLen(pts: { x: number; y: number; z: number }[]): number {
   630→  let L = 0;
   631→  for (let i = 1; i < pts.length; i++) {
   632→    L += Math.hypot(pts[i].x - pts[i - 1].x, pts[i].y - pts[i - 1].y, pts[i].z - pts[i - 1].z);
   633→  }
   634→  return L;
   635→}
   636→
   637→function buildMepRT(spec: MepSpec, tinggi: number): MepRT {
   638→  const routes: MepRouteRT[] = spec.routes.map((r) => {
   639→    const pts = r.pts.map((p) => ({ x: p.x, y: p.rel ? tinggi + p.h : p.h, z: p.z }));
   640→    return { id: r.id, jenis: r.jenis, label: r.label, sirkuit: r.sirkuit ?? null, pts, len: polylineLen(pts) };
   641→  });
   642→  const points: MepPointRT[] = spec.points.map((p) => ({
   643→    id: p.id,
   644→    kind: p.kind,
   645→    label: p.label,
   646→    x: p.x,
   647→    y: p.rel ? tinggi + p.h : p.h,
   648→    z: p.z,
   649→  }));
   650→  const byJenis = (j: MepJenis) => routes.filter((r) => r.jenis === j).reduce((s, r) => s + r.len, 0);
   651→  const cnt = (k: MepKind) => points.filter((p) => p.kind === k).length;
   652→  return {
   653→    airBersihM: byJenis("air_bersih"),
   654→    airKotorM: byJenis("air_kotor"),
   655→    conduitM: byJenis("listrik"),
   656→    kabelLampuM: routes.filter((r) => r.jenis === "listrik" && r.sirkuit !== "kontak").reduce((s, r) => s + r.len, 0),
   657→    kabelKontakM: routes.filter((r) => r.jenis === "listrik" && r.sirkuit === "kontak").reduce((s, r) => s + r.len, 0),
   658→    titikLampu: cnt("lampu"),
   659→    saklar: cnt("saklar"),
   660→    stopkontak: cnt("stopkontak"),
   661→    kloset: cnt("kloset"),
   662→    shower: cnt("shower"),
   663→    kran: cnt("kran"),
   664→    drain: cnt("drain"),
   665→    routes,
   666→    points,
   667→    septic: spec.septic,
   668→    resapan: spec.resapan,
   669→  };
   670→}

   674→  walls: WallRT[];
   675→  brickDims: { L: number; H: number };
   676→  bricks: BrickInstance[];
   677→  count: {
   678→    outerPieces: number;
   679→    innerPieces: number;
   680→    totalPieces: number;
   681→    buyOuter: number;
   682→    buyInner: number;
   683→    totalBuy: number;
   684→  };
   685→  mortar: {
   686→    volM3: number;
   687→    semenZak: number;
   688→    pasirM3: number;
   689→    perekatKg: number;
   690→    perekatZak: number;
   691→    natMm: number;
   692→  };
   693→  areas: {
   694→    gross: number;
   695→    netFloor: number;
   696→    rooms: RoomNet[];
   697→    reduksiM2: number;
   698→    reduksiPct: number;
   699→    wallFace: number;
   700→    plasterArea: number;
   701→    wallFootprint: number;
   702→  };
   703→  openings: { doors: number; windows: number; vents: number; windowAreaM2: number };
   704→  vent: { bukaanM2: number; pctLantai: number; okSni: boolean; okIdeal: boolean };
   705→  roof: {
   706→    areaM2: number;
   707→    apexAdd: number;
   708→    overhang: number;
   709→    deg: number;
   710→    pitch: number; // tan(deg)
   711→    yRidge: number; // ketinggian nok (m)
   712→    kudaKuda: number;
   713→    rengPerSlope: number;
   714→    gentengPcs: number;
   715→  };
   716→  wallLenTotal: number;
   717→  koloms: KolomRT[];
   718→  mep: MepRT;
   719→  structure: {
   720→    galianM3: number;
   721→    uruganPasirM3: number;
   722→    pondasiM3: number;
   723→    sloofM3: number;
   724→    kolomM3: number;
   725→    ringbalkM3: number;
   726→    lisplankM: number;
   727→    talangM: number;
   728→    cakar: CakarAyamRT | null;
   729→  };
   730→}
   731→
   732→export function buildModel(config: HouseConfig): HouseModel {
   733→  const preset = LAYOUT_PRESETS.find((p) => p.id === config.preset) ?? LAYOUT_PRESETS[0];
   734→  const tL = config.tLuar / 1000;
   735→  const tD = config.tDalam / 1000;
   736→  const dims = BRICK[config.jenis];
   737→  const j = config.natMm / 1000;
   738→
   739→  // geometri atap pelana: ridge memanjang arah Y denah (dinding ujung pelana = depan/belakang)
   740→  const rad = (config.atapDeg * Math.PI) / 180;
   741→  const apexAdd = 3 * Math.tan(rad); // setengah bentang 3 m -> naik ridge
   742→  const pitch = Math.tan(rad);
   743→  const rbTop = config.tinggi + RB_H; // puncak ring balk = dasar dinding pelana
   744→
   745→  // ----- dinding runtime
   746→  const walls: WallRT[] = preset.walls.map((spec) => {
   747→    const t = spec.outer ? tL : tD;
   748→    let c: number;
   749→    if (spec.edge !== undefined) {
   750→      c = spec.edge === 0 ? t / 2 : 6 - t / 2;
   751→    } else {
   752→      c = spec.pos ?? 0;
   753→    }
   754→    const lo = typeof spec.from === "function" ? spec.from({ tL, tD }) : spec.from;
   755→    const hi = typeof spec.to === "function" ? spec.to({ tL, tD }) : spec.to;
   756→    const explodeDir: [number, number] =
   757→      spec.axis === "h"
   758→        ? spec.edge === 6 || (!spec.outer && (spec.pos ?? 0) < 3)
   759→          ? [0, 1]
   760→          : [0, -1]
   761→        : spec.edge === 6 || (!spec.outer && (spec.pos ?? 0) > 3)
   762→          ? [1, 0]
   763→          : [-1, 0];
   764→    return {
   765→      spec,
   766→      id: spec.id,
   767→      name: spec.name,
   768→      axis: spec.axis,
   769→      c,
   770→      lo,
   771→      hi,
   772→      t,
   773→      h: config.tinggi,
   774→      gable: spec.gable ? { ...spec.gable, apexAdd, baseZ: rbTop } : undefined,
   775→      openings: spec.openings
   776→        .map((o) => resolveVent(o, { rbTop, apexAdd, tinggi: config.tinggi }))
   777→        .filter((o): o is OpeningSpec => !!o)
   778→        .filter((o) => o.to > lo && o.from < hi && o.top > 0.2),
   779→      outer: spec.outer,
   780→      explodeDir,
   781→    };
   782→  });
   783→
   784→  // ----- susunan bata per dinding
   785→  const bricks: BrickInstance[] = [];
   786→  let outerPieces = 0;
   787→  let innerPieces = 0;
   788→  let wallFace = 0;
   789→  let bandVol = 0;
   790→  let brickVol = 0;
   791→  let wallFootprint = 0;
   792→  let wallLenTotal = 0;
   793→
   794→  const cellL = dims.L + j;
   795→  const cellH = dims.H + j;
   796→
   797→  for (const w of walls) {
   798→    const len = w.hi - w.lo;
   799→    wallLenTotal += len;
   800→    wallFootprint += len * w.t;
   801→
   802→    const hAt = w.gable
   803→      ? (x: number) => w.gable!.baseZ + w.gable!.apexAdd * clamp01(1 - Math.abs(x - w.gable!.apexAt) / w.gable!.halfSpan)
   804→      : () => w.h;
   805→
   806→    // luas muka satu sisi (net bukaan: tinggi bukaan = top - sill, dipotong tinggi
   807→    // dinding datar / pelana; band ring balk tidak dihitung karena beton)
   808→    const gableArea = w.gable ? w.gable.apexAdd * (w.hi - w.lo) * 0.5 : 0;
   809→    const hAtMid = (o: OpeningSpec) =>
   810→      Math.max(w.h, hAt((o.from + o.to) / 2));
   811→    const openingArea = w.openings.reduce(
   812→      (s, o) =>
   813→        s +
   814→        Math.max(0, Math.min(o.to, w.hi) - Math.max(o.from, w.lo)) *
   815→          Math.max(0, Math.min(o.top, hAtMid(o)) - o.sill),
   816→      0,
   817→    );
   818→    wallFace += w.h * len + gableArea - openingArea;
   819→    bandVol += (w.h * len + gableArea) * w.t - openingArea * w.t;
   820→
   821→    const topZ = w.gable ? w.gable.baseZ + w.gable.apexAdd : w.h;
   822→    const nCourses = Math.ceil(topZ / cellH);
   823→    for (let i = 0; i < nCourses; i++) {
   824→      const zc = i * cellH;
   825→      if (zc >= topZ - 0.001) break;
   826→
   827→      // interval yang boleh dipasangi bata pada lajur ini:
   828→      // 1) band ring balk (w.h .. baseZ) tidak dipasangi bata; di atas baseZ
   829→      //    bata mengikuti penyempitan segitiga pelana
   830→      if (w.gable && zc >= w.h && zc < w.gable.baseZ) continue; // zona ring balk
   831→
   832→      const allowed: Interval = { lo: w.lo, hi: w.hi };
   833→      if (w.gable && zc >= w.gable.baseZ) {
   834→        const d = w.gable.halfSpan * (1 - (zc - w.gable.baseZ) / w.gable.apexAdd);
   835→        if (d <= 0) continue;
   836→        allowed.lo = Math.max(w.lo, w.gable.apexAt - d);
   837→        allowed.hi = Math.min(w.hi, w.gable.apexAt + d);
   838→      }
   839→      if (allowed.hi - allowed.lo <= 0.015) continue;
   840→
   841→      // 2) bukaan yang memotong lajur ini secara vertikal
   842→      const activeCuts: Interval[] = w.openings
   843→        .filter((o) => o.sill < zc + dims.H - 0.005 && o.top > zc + 0.005)
   844→        .map((o) => ({ lo: Math.max(o.from, w.lo), hi: Math.min(o.to, w.hi) }));
   845→
   846→      // 3) isi bata dengan ikatan berselang (stretcher bond)
   847→      const phase = (i % 2) * cellL * 0.5;
   848→      let p = allowed.lo - phase;
   849→      while (p + dims.L <= allowed.lo) p += cellL;
   850→      for (; p < allowed.hi; p += cellL) {
   851→        const segs = subtractIntervals(
   852→          { lo: Math.max(p, allowed.lo), hi: Math.min(p + dims.L, allowed.hi) },
   853→          activeCuts,
   854→        );
   855→        for (const s of segs) {
   856→          const cx = (s.lo + s.hi) / 2;
   857→          // di bawah ring balk dipotong rata di tinggi dinding; di atas mengikuti kemiringan pelana
   858→          const hgt =
   859→            zc < w.h
   860→              ? Math.min(dims.H, Math.max(0, w.h - zc))
   861→              : Math.min(dims.H, Math.max(0, hAt(cx) - zc));
   862→          if (hgt <= 0.02) continue;
   863→          const frac = ((s.hi - s.lo) / dims.L) * (hgt / dims.H);
   864→          if (w.outer) outerPieces += frac;
   865→          else innerPieces += frac;
   866→          brickVol += frac * dims.L * dims.H * w.t;
   867→          bricks.push(
   868→            w.axis === "h"
   869→              ? { wallId: w.id, outer: w.outer, x: cx, y: w.c, z: zc, len: s.hi - s.lo, hgt, rotY: 0 }
   870→              : { wallId: w.id, outer: w.outer, x: w.c, y: cx, z: zc, len: s.hi - s.lo, hgt, rotY: Math.PI / 2 },
   871→          );
   872→        }
   873→      }
   874→    }
   875→  }
   876→
   877→  const mortarVol = Math.max(0, bandVol - brickVol);
   878→
   879→  // ----- luas bersih ruangan (setelah dikurangi tebal dinding)
   880→  interface Band {
   881→    axis: Axis;
   882→    x0: number;
   883→    x1: number;
   884→    y0: number;
   885→    y1: number;
   886→  }
   887→  const bands: Band[] = walls.map((w) =>
   888→    w.axis === "h"
   889→      ? { axis: w.axis, x0: w.lo, x1: w.hi, y0: w.c - w.t / 2, y1: w.c + w.t / 2 }
   890→      : { axis: w.axis, x0: w.c - w.t / 2, x1: w.c + w.t / 2, y0: w.lo, y1: w.hi },
   891→  );
   892→  const rooms: RoomNet[] = preset.rooms.map((r) => {
   893→    let xl = 0, xr = 0, yb = 0, yt = 0;
   894→    for (const b of bands) {
   895→      const yOverlap = Math.min(b.y1, r.y1) - Math.max(b.y0, r.y0);
   896→      const xOverlap = Math.min(b.x1, r.x1) - Math.max(b.x0, r.x0);
   897→      if (b.axis === "v") {
   898→        if (b.x0 <= r.x0 + 0.02 && b.x1 > r.x0 && yOverlap > 0.05) xl = Math.max(xl, b.x1 - r.x0);
   899→        if (b.x1 >= r.x1 - 0.02 && b.x0 < r.x1 && yOverlap > 0.05) xr = Math.max(xr, r.x1 - b.x0);
   900→      } else {
   901→        if (b.y0 <= r.y0 + 0.02 && b.y1 > r.y0 && xOverlap > 0.05) yb = Math.max(yb, b.y1 - r.y0);
   902→        if (b.y1 >= r.y1 - 0.02 && b.y0 < r.y1 && xOverlap > 0.05) yt = Math.max(yt, r.y1 - b.y0);
   903→      }
   904→    }
   905→    const wNet = Math.max(0, r.x1 - r.x0 - xl - xr);
   906→    const dNet = Math.max(0, r.y1 - r.y0 - yb - yt);
   907→    return { id: r.id, name: r.name, w: wNet, d: dNet, area: wNet * dNet, cx: (r.x0 + r.x1) / 2, cy: (r.y0 + r.y1) / 2 };
   908→  });
   909→
   910→  const netFloor = rooms.reduce((s, r) => s + r.area, 0);
   911→  const gross = 36;
   912→
   913→  // ----- kolom praktis 15/15: pada tiap pertemuan centerline dinding h x v
   914→  const seen = new Set<string>();
   915→  const koloms: KolomRT[] = [];
   916→  const tol = 0.15;
   917→  for (const h of walls) {
   918→    if (h.axis !== "h") continue;
   919→    for (const v of walls) {
   920→      if (v.axis !== "v") continue;
   921→      if (v.c >= h.lo - tol && v.c <= h.hi + tol && h.c >= v.lo - tol && h.c <= v.hi + tol) {
   922→        const key = `${v.c.toFixed(2)}|${h.c.toFixed(2)}`;
   923→        if (!seen.has(key)) {
   924→          seen.add(key);
   925→          koloms.push({ x: v.c, y: h.c });
   926→        }
   927→      }
   928→    }
   929→  }
   930→
   931→  // ----- material
   932→  const waste = config.wastePct / 100;
   933→  const mortarWithWaste = mortarVol * (1 + waste);
   934→  const semenZak = config.jenis === "merah" ? mortarWithWaste * 6.3 : 0;
   935→  const pasirM3 = config.jenis === "merah" ? mortarWithWaste * 1.0 : 0;
   936→  const perekatKg = config.jenis === "hebel" ? wallFace * 4 : 0;
   937→  const perekatZak = config.jenis === "hebel" ? perekatKg / 40 : 0;
   938→
   939→  // ----- atap pelana (kuda-kuda C75 jarak maks 1,2 m, genteng beton 10 pcs/m2)
   940→  const ov = config.overhang;
   941→  const yRidge = rbTop + apexAdd;
   942→  const slopeHalf = Math.hypot(3 + ov, apexAdd + ov * pitch);
   943→  const roofArea = 2 * slopeHalf * (6 + 2 * ov);
   944→  // jumlah kuda-kuda: seluruh bentang 6 m dengan jarak antar rangka maks 1,2 m
   945→  const kudaKuda = Math.max(4, Math.ceil(6 / KUDA_SPACING) + 1);
   946→  const rengPerSlope = Math.ceil(3 / 0.4) + 1;
   947→  const gentengPcs = Math.ceil(roofArea * 10);
   948→
   949→  const doors = walls.reduce((s, w) => s + w.openings.filter((o) => o.kind === "pintu").length, 0);
   950→  const windows = walls.reduce((s, w) => s + w.openings.filter((o) => o.kind === "jendela").length, 0);
   951→  const vents = walls.reduce((s, w) => s + w.openings.filter((o) => o.kind === "vent").length, 0);
   952→  const windowAreaM2 = walls.reduce(
   953→    (s, w) => s + w.openings.filter((o) => o.kind === "jendela").reduce((a, o) => a + (o.to - o.from) * (o.top - o.sill), 0),
   954→    0,
   955→  );
   956→
   957→  // ----- pondasi cakar ayam: buis di tiap titik kolom + grid tengah (jarak maks ~3 m)
   958→  let cakar: CakarAyamRT | null = null;
   959→  if (config.pondasi === "cakarayam") {
   960→    const pts: { x: number; y: number }[] = koloms.map((k) => ({ x: k.x, y: k.y }));
   961→    for (const gx of [1.5, 4.5]) for (const gy of [1.5, 4.5]) pts.push({ x: gx, y: gy });
   962→    const buisPts: { x: number; y: number }[] = [];
   963→    for (const p of pts) {
   964→      if (!buisPts.some((q) => Math.hypot(q.x - p.x, q.y - p.y) < 0.45)) buisPts.push(p);
   965→    }
   966→    const rOut = CAKAR.buisDia / 2;
   967→    const rIn = rOut - CAKAR.buisWall;
   968→    const buisM3 = buisPts.length * Math.PI * (rOut * rOut - rIn * rIn) * CAKAR.buisL;
   969→    const slabM3 = 36 * CAKAR.slabT;
   970→    cakar = {
   971→      slabM3,
   972→      buisCount: buisPts.length,
   973→      buisM3,
   974→      uruganM3: 36 * CAKAR.bed,
   975→      galianM3: 36 * CAKAR.bed + buisPts.length * Math.PI * rOut * rOut * CAKAR.buisL,
   976→      betonM3: slabM3 + buisM3,
   977→      buisPts,
   978→    };
   979→  }
   980→
   981→  // ----- kuantitas struktur (galian & pondasi mengikuti jenis pondasi)
   982→  const galianM3 =
   983→    config.pondasi === "cakarayam" && cakar
   984→      ? cakar.galianM3
   985→      : wallLenTotal * PONDASI_D * GALIAN_W; // galian 60 x 60 cm
   986→  const uruganPasirM3 = config.pondasi === "cakarayam" ? 0 : wallLenTotal * GALIAN_W * URUGAN_T;
   987→  const pondasiM3 = config.pondasi === "cakarayam" ? 0 : wallLenTotal * PONDASI_W * PONDASI_D;
   988→  const sloofM3 = wallLenTotal * SLOOF_W * RB_H;
   989→  const kolomM3 = koloms.length * KOLOM_S * KOLOM_S * (config.tinggi + RB_H);
   990→  const ringbalkM3 = wallLenTotal * SLOOF_W * RB_H;
   991→  const lisplankM = 2 * (6 + 2 * ov) + 2 * slopeHalf; // eaves + kedua ujung pelana
   992→  const talangM = 2 * (6 + 2 * ov);
   993→
   994→  return {
   995→    config,
   996→    walls,
   997→    brickDims: { L: dims.L, H: dims.H },
   998→    bricks,
   999→    count: {
  1000→      outerPieces,
  1001→      innerPieces,
  1002→      totalPieces: outerPieces + innerPieces,
  1003→      buyOuter: Math.ceil(outerPieces * (1 + waste)),
  1004→      buyInner: Math.ceil(innerPieces * (1 + waste)),
  1005→      totalBuy: Math.ceil(outerPieces * (1 + waste)) + Math.ceil(innerPieces * (1 + waste)),
  1006→    },
  1007→    mortar: {
  1008→      volM3: mortarVol,
  1009→      semenZak,
  1010→      pasirM3,
  1011→      perekatKg,
  1012→      perekatZak,
  1013→      natMm: config.natMm,
  1014→    },
  1015→    areas: {
  1016→      gross,
  1017→      netFloor,
  1018→      rooms,
  1019→      reduksiM2: gross - netFloor,
  1020→      reduksiPct: ((gross - netFloor) / gross) * 100,
  1021→      wallFace,
  1022→      plasterArea: wallFace * 2,
  1023→      wallFootprint,
  1024→    },
  1025→    openings: { doors, windows, vents, windowAreaM2 },
  1026→    vent: {
  1027→      bukaanM2: windowAreaM2,
  1028→      pctLantai: (windowAreaM2 / Math.max(0.01, netFloor)) * 100,
  1029→      okSni: windowAreaM2 >= 0.05 * netFloor,
  1030→      okIdeal: windowAreaM2 >= 0.1 * netFloor,
  1031→    },
  1032→    roof: { areaM2: roofArea, apexAdd, overhang: ov, deg: config.atapDeg, pitch, yRidge, kudaKuda, rengPerSlope, gentengPcs },
  1033→    wallLenTotal,
  1034→    koloms,
  1035→    mep: buildMepRT(preset.mep, config.tinggi),

  1128→export function buildRab(model: HouseModel, prices: PriceBook, upahPct: number): RabSummary {
  1129→  const cfg = model.config;
  1130→  const rows: RabRow[] = [];
  1131→  const push = (id: string, divisi: string, uraian: string, spek: string, qty: number, unit: string, exact: boolean, priceKey?: string) => {
  1132→    const key = priceKey ?? id;
  1133→    const harga = prices[key] ?? 0;
  1134→    rows.push({ id, divisi, uraian, spek, qty, unit, harga, subtotal: qty * harga, exact, priceKey: key });
  1135→  };
  1136→
  1137→  push("bouwplank", "d1", "Pengukuran & bouwplank", "Profil kayu + patok as bangunan", 1, "lot", false);
  1138→
  1139→  if (cfg.pondasi === "cakarayam" && model.structure.cakar) {
  1140→    const c = model.structure.cakar;
  1141→    push("galian", "d2", "Galian titik buis + lantai kerja pelat", "Lubang buis Ø40 cm sedalam 80 cm + perataan 5 cm di seluruh pelat", Math.ceil(c.galianM3 * 10) / 10, "m3", true);
  1142→    push("urugan", "d2", "Urugan pasir bawah pelat", "Pasir setempat 5 cm, dipadatkan sebelum cor pelat", Math.ceil(c.uruganM3 * 10) / 10, "m3", true, "urugan");
  1143→    push("cakarBeton", "d2", "Pondasi cakar ayam (sistem Sedyatmo)", `Pelat 6x6 m tebal 10 cm wiremesh D8 + ${c.buisCount} buis beton Ø40 cm panjang 80 cm, cor monolit`, Math.ceil(c.betonM3 * 100) / 100, "m3", true, "cakarBeton");
  1144→  } else {
  1145→    push("galian", "d2", "Galian tanah pondasi", "Galian 60 x 60 cm sedalam 60 cm sepanjang jalur dinding", Math.ceil(model.structure.galianM3 * 10) / 10, "m3", true);
  1146→    push("uruganPasir", "d2", "Urugan pasir bawah pondasi", "Pasir setempat 10 cm dipadatkan di dasar galian", Math.ceil(model.structure.uruganPasirM3 * 10) / 10, "m3", true, "urugan");
  1147→    push("pondasi", "d2", "Pondasi batu kali + campuran 1:5", "Strip 40 x 60 cm (kedalaman 60 cm)", Math.ceil(model.structure.pondasiM3 * 10) / 10, "m3", true);
  1148→  }
  1149→
  1150→  push("sloof", "d3", "Sloof beton bertulang 15/20", "Beton K225, tulangan 6D10, begel Ø8-150", Math.ceil(model.structure.sloofM3 * 100) / 100, "m3", true);
  1151→  push("kolom", "d3", "Kolom praktis beton 15/15", "4D10, begel Ø6-200, jarak maks 3 m", Math.ceil(model.structure.kolomM3 * 100) / 100, "m3", true);
  1152→  push("ringbalk", "d3", "Ring balk beton bertulang 15/20", "Mengikat seluruh dinding di ketinggian 3 m", Math.ceil(model.structure.ringbalkM3 * 100) / 100, "m3", true);
  1153→

  1155→    push("bataLuar", "d4", `Bata putih (hebel) dinding luar ${cfg.tLuar} mm`, "60 x 20 cm SNI 8640, dihitung dari susunan 3D + waste", model.count.buyOuter, "buah", true, hebelPriceId(cfg.tLuar));
  1156→    push("bataDalam", "d4", `Bata putih (hebel) dinding dalam ${cfg.tDalam} mm`, "60 x 20 cm SNI 8640, dihitung dari susunan 3D + waste", model.count.buyInner, "buah", true, hebelPriceId(cfg.tDalam));
  1157→    push("perekat", "d4", "Perekat hebel (mortar instan)", "4 kg per m2 dinding, nat 2-3 mm, kemasan 40 kg/zak", Math.ceil(model.mortar.perekatZak), "zak", true);
  1158→  } else {
  1159→    push("bataLuar", "d4", "Bata merah dinding luar (1/2 batu)", "20x10x5 cm, dihitung dari susunan 3D + waste", model.count.buyOuter, "buah", true, "bataMerah");
  1160→    push("bataDalam", "d4", "Bata merah dinding dalam (1/2 batu)", "20x10x5 cm, dihitung dari susunan 3D + waste", model.count.buyInner, "buah", true, "bataMerah");
  1161→    push("semen", "d4", "Semen pasangan (campuran 1:5)", `Nat ${cfg.natMm} mm; volume adukan ${model.mortar.volM3.toFixed(2)} m3`, Math.ceil(model.mortar.semenZak), "zak", true);
  1162→    push("pasir", "d4", "Pasir pasang", "Campuran adukan 1:5", Math.ceil(model.mortar.pasirM3 * 10) / 10, "m3", true);

  1164→
  1165→  if (cfg.plesteran) {
  1166→    push("plesteran", "d4", "Plesteran + aci (2 sisi dinding)", "Campuran 1:5 tebal 10-15 mm", Math.ceil(model.areas.plasterArea * 10) / 10, "m2", true);
  1167→  }

  1172→  push("genteng", "d5", "Genteng beton flat + pasang", `10 pcs/m2 (${model.roof.gentengPcs} pcs), kemiringan ${cfg.atapDeg} derajat`, Math.ceil(model.roof.areaM2), "m2", true);
  1173→  push("lisplank", "d5", "Lisplank", "Papan penutup ujung rangka atap (eaves + kedua ujung pelana)", Math.ceil(model.structure.lisplankM * 10) / 10, "m1", true);

  1175→  push("plafon", "d5", "Plafon gypsum + rangka hollow galvalum", "Gypsum 9 mm di atas luas bersih lantai", Math.ceil(model.areas.netFloor * 10) / 10, "m2", false);
  1176→
  1177→  push("rabat", "d6", "Rabat beton lantai kerja", "Tebal 10 cm di atas luas bersih lantai", Math.ceil(model.areas.netFloor * 10) / 10, "m2", false);
  1178→  push("keramik", "d6", "Keramik lantai + pemasangan", "40x40 cm golongan menengah", Math.ceil(model.areas.netFloor * 10) / 10, "m2", false);
  1179→
  1180→  push("pintu", "d7", "Kusen + daun pintu lengkap", "Kusen kayu berprofil + daun panel 2 susun + kunci handle + angkur", model.openings.doors, "unit", false);
  1181→  push("jendela", "d7", "Kusen + kaca jendela", "Kusen aluminium profil + kaca 5 mm + palang tengah + roster vent", model.openings.windows, "unit", false);
  1182→
  1183→  push("listrikAir", "d8", "Instalasi listrik & air bersih/limbah", "Titik lampu + stop kontak, MCB, pipa air & kotoran", 1, "borongan", false);
  1184→
  1185→  push("cat", "d9", "Cat dinding (primer + 2 lapis)", "Di atas luas aci 2 sisi dinding", Math.ceil(model.areas.plasterArea * 10) / 10, "m2", true);
  1186→  push("pembersihan", "d9", "Pembersihan akhir & perapihan", "Serah terima pekerjaan", 1, "lot", false);
  1187→
  1188→  const divisions: RabDivision[] = DIVISIONS.map((d) => {
  1189→    const drows = rows.filter((r) => r.divisi === d.id);
  1190→    return {
  1191→      id: d.id,
  1192→      kode: d.kode,
  1193→      nama: d.nama,
  1194→      rows: drows,
  1195→      subtotal: drows.reduce((s, r) => s + r.subtotal, 0),
  1196→    };
  1197→  }).filter((d) => d.rows.length > 0);
  1198→
  1199→  const materialTotal = rows.reduce((s, r) => s + r.subtotal, 0);
  1200→  const upah = Math.round((materialTotal * upahPct) / 100);
  1201→  const total = materialTotal + upah;
  1202→  return {
  1203→    rows,
  1204→    divisions,
  1205→    materialTotal,
  1206→    upahPct,
  1207→    upah,
  1208→    total,
  1209→    perM2Gross: total / model.areas.gross,
  1210→    perM2Net: total / Math.max(0.01, model.areas.netFloor),
  1211→  };
  1212→}
