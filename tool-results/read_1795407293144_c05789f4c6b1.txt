   214→  private initOrbit() {
   215→    this.canvas.style.touchAction = "none";
   216→    this.canvas.addEventListener("pointerdown", this.onDown);
   217→    this.canvas.addEventListener("pointermove", this.onMove);
   218→    this.canvas.addEventListener("pointerup", this.onUp);
   219→    this.canvas.addEventListener("pointercancel", this.onUp);
   220→    this.canvas.addEventListener("wheel", this.onWheel, { passive: false });
   221→    this.canvas.addEventListener("contextmenu", (e) => e.preventDefault());
   222→  }
   223→
   224→  private onDown = (e: PointerEvent) => {
   225→    this.canvas.setPointerCapture(e.pointerId);
   226→    this.pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
   227→    this.panMode = e.button === 1 || e.button === 2 || e.shiftKey;
   228→    if (this.pointers.size === 2) {
   229→      const pts = [...this.pointers.values()];
   230→      this.pinchDist = Math.hypot(pts[0].x - pts[1].x, pts[0].y - pts[1].y);
   231→    }
   232→  };
   233→
   234→  private onMove = (e: PointerEvent) => {
   235→    const prev = this.pointers.get(e.pointerId);
   236→    if (!prev) return;
   237→    const dx = e.clientX - prev.x;
   238→    const dy = e.clientY - prev.y;
   239→    this.pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
   240→
   241→    if (this.pointers.size === 2) {
   242→      const pts = [...this.pointers.values()];
   243→      const d = Math.hypot(pts[0].x - pts[1].x, pts[0].y - pts[1].y);
   244→      if (this.pinchDist > 0) this.sph.radius = THREE.MathUtils.clamp(this.sph.radius * (this.pinchDist / d), 3, 45);
   245→      this.pinchDist = d;
   246→      return;
   247→    }
   248→    if (this.panMode) this.pan(dx, dy);
   249→    else {
   250→      this.sph.theta -= dx * 0.0055;
   251→      this.sph.phi = THREE.MathUtils.clamp(this.sph.phi - dy * 0.0045, 0.06, Math.PI * 0.495);
   252→    }
   253→  };
   254→
   255→  private onUp = (e: PointerEvent) => {
   256→    this.pointers.delete(e.pointerId);
   257→    if (this.pointers.size < 2) this.pinchDist = 0;
   258→  };
   259→
   260→  private onWheel = (e: WheelEvent) => {
   261→    e.preventDefault();
   262→    this.sph.radius = THREE.MathUtils.clamp(this.sph.radius * (1 + e.deltaY * 0.0011), 3, 45);
   263→  };
   264→
   265→  private pan(dx: number, dy: number) {
   266→    const fwd = new THREE.Vector3().subVectors(this.target, this.camPos()).setY(0).normalize();
   267→    const right = new THREE.Vector3().crossVectors(new THREE.Vector3(0, 1, 0), fwd).normalize();
   268→    const k = this.sph.radius * 0.0012;
   269→    this.target.addScaledVector(right, -dx * k);
   270→    this.target.addScaledVector(fwd, -dy * k);
   271→    this.target.x = THREE.MathUtils.clamp(this.target.x, -6, 12);
   272→    this.target.z = THREE.MathUtils.clamp(this.target.z, -6, 12);
   273→    this.target.y = 1.2;
   274→  }

   276→  setCameraPreset(p: CameraPreset) {
   277→    if (p === "depan") {
   278→      this.target.set(3, 2.0, 3);
   279→      this.sph = { radius: 12.5, theta: 0, phi: 1.3 };
   280→    } else if (p === "belakang") {
   281→      this.target.set(3, 2.0, 3);
   282→      this.sph = { radius: 12.5, theta: Math.PI, phi: 1.3 };
   283→    } else if (p === "samping") {
   284→      this.target.set(3, 2.0, 3);
   285→      this.sph = { radius: 12.5, theta: Math.PI / 2, phi: 1.3 };
   286→    } else if (p === "atas") {
   287→      this.target.set(3, 0, 3);
   288→      this.sph = { radius: 13.5, theta: Math.PI * 0.001, phi: 0.03 };
   289→    } else {
   290→      this.target.set(3, 1.4, 3);
   291→      this.sph = { radius: 13, theta: Math.PI * 0.15, phi: 1.02 };
   292→    }
   293→  }

   320→  private camPos(): THREE.Vector3 {
   321→    const { radius, theta, phi } = this.sph;
   322→    return new THREE.Vector3(
   323→      this.target.x + radius * Math.sin(phi) * Math.sin(theta),
   324→      this.target.y + radius * Math.cos(phi),
   325→      this.target.z + radius * Math.sin(phi) * Math.cos(theta),
   326→    );
   327→  }

   329→  private loop = () => {
   330→    this.raf = requestAnimationFrame(this.loop);
   331→    const cam: THREE.PerspectiveCamera | THREE.OrthographicCamera = this.useOrtho ? this.cameraOrtho : this.camera;
   332→    cam.position.copy(this.camPos());
   333→    cam.lookAt(this.target);
   334→    if (this.useOrtho) this.syncOrtho();
   335→    this.renderer.render(this.scene, cam);
   336→  };

   338→  private syncOrtho() {
   339→    const el = this.canvas.parentElement ?? this.canvas;
   340→    const w = Math.max(1, el.clientWidth);
   341→    const h = Math.max(1, el.clientHeight);
   342→    const halfH = this.sph.radius * Math.tan((this.camera.fov * Math.PI) / 360);
   343→    const halfW = halfH * (w / h);
   344→    const o = this.cameraOrtho;
   345→    o.left = -halfW;
   346→    o.right = halfW;
   347→    o.top = halfH;
   348→    o.bottom = -halfH;
   349→    o.updateProjectionMatrix();
   350→  }
   351→
   352→  private resize() {
   353→    const el = this.canvas.parentElement ?? this.canvas;
   354→    const w = Math.max(1, el.clientWidth);
   355→    const h = Math.max(1, el.clientHeight);
   356→    this.renderer.setSize(w, h, false);
   357→    this.camera.aspect = w / h;
   358→    this.camera.updateProjectionMatrix();
   359→    this.syncOrtho();
   360→  }

   364→  update(model: HouseModel, view: ViewState) {
   365→    this.lastModel = model;
   366→    // bersihkan rumah lama (grup persisten dilepas, lalu dipasang kembali)
   367→    this.clearGroup(this.houseRoot);
   368→    this.houseRoot.add(this.kolomGroup);
   369→    this.clearGroup(this.kolomGroup);
   370→    this.clearGroup(this.roofGroup);
   371→    this.roofGroup.add(this.tilesGroup);
   372→    this.roofGroup.add(this.trussGroup);
   373→    this.clearGroup(this.tilesGroup);
   374→    this.clearGroup(this.trussGroup);
   375→    this.clearGroup(this.labelGroup);
   376→
   377→    // potongan
   378→    this.clipPlane.constant = view.cutH;
   379→    // warna cat eksterior mengikuti pilihan pengguna
   380→    this.matCatExt.color.set(view.paintHex || "#f2ecdf");
   381→    // pada tahap pondasi/sloof, tanah dibuat tembus pandang agar profil pondasi
   382→    // (batu kali / buis cakar ayam) di bawah muka tanah ikut terlihat
   383→    const semi = s2(view.stage) === 1 || s2(view.stage) === 2;
   384→
   385→    const isHebel = model.config.jenis === "hebel";
   386→    this.matBrick.color.set(isHebel ? 0xeceae3 : 0xa3492f);
   387→    // tahap pondasi/sloof: tanah dibuat tembus pandang & petak 6x6 disembunyikan
   388→    // agar profil pondasi (batu kali / buis cakar ayam) di bawah muka tanah terbaca
   389→    this.matGround.transparent = semi;
   390→    this.matGround.opacity = semi ? 0.45 : 1;
   391→    if (this.plotMesh) this.plotMesh.visible = this.envVisible.plot && !semi;
   392→    this.groundMesh.visible = this.envVisible.ground;
   393→    this.gridHelper.visible = this.envVisible.grid;
   394→
   395→    const tL = model.config.tLuar / 1000;
   396→    const tinggi = model.config.tinggi;
   397→    const cakar = model.config.pondasi === "cakarayam" && !!model.structure.cakar;
   398→
   399→    // ----- tahapan konstruksi: 0 = kendali bebas, 1..8 = kondisi pembangunan
   400→    const s = Math.max(0, Math.round(view.stage || 0));
   401→    const manual = s === 0;
   402→    const V: StageVis = {
   403→      pondasi: manual ? true : s >= 1,
   404→      sloof: manual ? view.showStruktur : s >= 2,
   405→      kolom: manual ? view.showStruktur : s >= 2,
   406→      dinding: manual ? view.showDinding : s >= 3,
   407→      pelana: manual ? view.showDinding : s >= 4,
   408→      ringbalk: manual ? view.showStruktur : s >= 4,
   409→      kusen: manual ? view.showDinding || view.showStruktur : s >= 3,
   410→      lantai: manual ? true : s >= 8,
   411→      rangka: manual ? view.showRangka : s >= 5,
   412→      genteng: manual ? view.showRoof : s >= 6,
   413→      plafon: manual ? view.showRoof || view.showRangka : s >= 6,
   414→      plester: manual ? view.showFinishing : s >= 7,
   415→      cat: manual ? view.showFinishing : s >= 8,
   416→    };
   417→
   418→    // lantai interior (rabat + keramik) di Y=0
   419→    if (V.lantai) {
   420→      const lantaiW = 6 - 2 * tL;
   421→      this.addBox(this.houseRoot, 3, -0.05, 3, lantaiW, 0.1, lantaiW, this.matLantai, true);
   422→    }
   423→
   424→    // pondasi cakar ayam (pelat + buis, sekali untuk seluruh denah)
   425→    if (cakar && V.pondasi && model.structure.cakar) {
   426→      this.buildCakarAyam(model.structure.cakar);
   427→    }
   428→
   429→    // dinding: pondasi batu kali + sloof + bata (instancing per dinding) + ring balk + kusen
   430→    for (const w of model.walls) {
   431→      this.buildWall(w, model, V, cakar);
   432→    }
   433→
   434→    // kolom praktis 15/15 di tiap simpul dinding
   435→    if (V.kolom) {
   436→      const kw = Math.max(0.15, tL + 0.015);
   437→      for (const k of model.koloms) {
   438→        this.addBox(this.kolomGroup, k.x, (tinggi + RB_H) / 2, k.y, kw, tinggi + RB_H, kw, this.matBeton);
   439→      }
   440→    }
   441→
   442→    // atap + plafon
   443→    this.buildRoof(model, V);
   444→
   445→    // label ruangan
   446→    if (view.showLabels) {
   447→      for (const room of model.areas.rooms) {
   448→        if (room.area <= 0.5) continue;
   449→        const sp = this.makeTextSprite(room.name, `${room.w.toFixed(2)} x ${room.d.toFixed(2)} m | ${room.area.toFixed(1)} m2`);
   450→        sp.position.set(room.cx, 1.55, room.cy);
   451→        this.labelGroup.add(sp);
   452→      }
   453→    }
   454→    this.labelGroup.visible = view.showLabels;
   455→
   456→    // mode bongkar: atap naik, dinding mengurai keluar
   457→    this.roofGroup.position.y = view.explode * 2.6;
   458→    for (const g of this.houseRoot.children) {
   459→      if (g instanceof THREE.Group && g.userData.explodeDir) {
   460→        const d = g.userData.explodeDir as [number, number];
   461→        g.position.set(d[0] * view.explode * 1.4, 0, d[1] * view.explode * 1.4);
   462→      }
   463→    }
   464→    this.roofGroup.visible = manual
   465→      ? view.showRoof || view.showRangka
   466→      : V.rangka || V.genteng || V.plafon;
   467→    this.tilesGroup.visible = V.genteng;
   468→    this.trussGroup.visible = V.rangka;
   469→    this.kolomGroup.visible = V.kolom;
   470→
   471→    // anotasi dimensi 3D mengikuti model terakhir
   472→    this.rebuildDims();
   473→  }

   475→  private buildWall(w: WallRT, model: HouseModel, V: StageVis, cakar: boolean) {
   476→    const group = new THREE.Group();
   477→    group.userData.explodeDir = w.explodeDir;
   478→    this.houseRoot.add(group);
   479→
   480→    const len = w.hi - w.lo;
   481→    const t = w.t;
   482→    const mid = (w.lo + w.hi) / 2;
   483→    const bw = Math.max(0.15, t + 0.01); // lebar sloof / ring balk (15 cm standar)
   484→
   485→    // pondasi batu kali: 40 x 60 cm, dari -0,6 s/d -0,2 m (digantikan cakar ayam bila aktif)
   486→    // sloof beton 15/20: dari -0,2 s/d 0 m | ring balk: dari tinggi s/d tinggi + 0,2 m
   487→    if (!cakar && V.pondasi) {
   488→      const pt = t + 0.15;
   489→      if (w.axis === "h") this.addBox(group, mid, -0.4, w.c, len, 0.4, pt, this.matPlinth);
   490→      else this.addBox(group, w.c, -0.4, mid, pt, 0.4, len, this.matPlinth);
   491→    }
   492→    if (V.sloof) {
   493→      if (w.axis === "h") this.addBox(group, mid, -0.1, w.c, len, RB_H, bw, this.matBeton);
   494→      else this.addBox(group, w.c, -0.1, mid, bw, RB_H, len, this.matBeton);
   495→    }
   496→    if (V.ringbalk) {
   497→      if (w.axis === "h") this.addBox(group, mid, w.h + RB_H / 2, w.c, len, RB_H, bw, this.matBeton);
   498→      else this.addBox(group, w.c, w.h + RB_H / 2, mid, bw, RB_H, len, this.matBeton);
   499→    }
   500→
   501→    // mortar/nat slab: kotak-kotak menghindari bukaan (hanya sampai tinggi dinding datar)
   502→    const segs: { lo: number; hi: number; z0: number; z1: number }[] = [];
   503→    const ops = [...w.openings].sort((a, b) => a.from - b.from);
   504→    let cursor = w.lo;
   505→    for (const o of ops) {
   506→      const clo = Math.max(o.from, w.lo);
   507→      const chi = Math.min(o.to, w.hi);
   508→      if (clo > cursor) segs.push({ lo: cursor, hi: clo, z0: 0, z1: w.h });
   509→      if (o.sill > 0) segs.push({ lo: clo, hi: chi, z0: 0, z1: o.sill });
   510→      if (o.top < w.h) segs.push({ lo: clo, hi: chi, z0: o.top, z1: w.h });
   511→      cursor = Math.max(cursor, chi);
   512→    }
   513→    if (cursor < w.hi) segs.push({ lo: cursor, hi: w.hi, z0: 0, z1: w.h });
   514→
   515→    // mortar/nat terlihat di sela bata — tertutup begitu plester dipasang
   516→    if (V.dinding && !V.plester) {
   517→      for (const s of segs) {
   518→        if (s.hi - s.lo < 0.01 || s.z1 - s.z0 < 0.01) continue;
   519→        if (w.axis === "h") this.addBox(group, (s.lo + s.hi) / 2, (s.z0 + s.z1) / 2, w.c, s.hi - s.lo, s.z1 - s.z0, t, this.matMortar);
   520→        else this.addBox(group, w.c, (s.z0 + s.z1) / 2, (s.lo + s.hi) / 2, t, s.z1 - s.z0, s.hi - s.lo, this.matMortar);
   521→      }
   522→    }
   523→
   524→    // segitiga mortar di ujung atap pelana (dinding depan/belakang), di atas ring balk
   525→    if (w.gable && V.pelana && !V.plester) {
   526→      const baseZ = w.gable.baseZ;
   527→      const shape = new THREE.Shape();
   528→      shape.moveTo(w.lo, baseZ);
   529→      shape.lineTo(w.hi, baseZ);
   530→      shape.lineTo(w.gable.apexAt, baseZ + w.gable.apexAdd);
   531→      shape.closePath();
   532→      const geo = new THREE.ExtrudeGeometry(shape, { depth: t, bevelEnabled: false });
   533→      const mesh = new THREE.Mesh(geo, this.matMortar);
   534→      mesh.castShadow = true;
   535→      mesh.receiveShadow = true;
   536→      mesh.position.set(0, 0, w.c - t / 2);
   537→      group.add(mesh);
   538→      this.disposables.push(geo);
   539→    }
   540→
   541→    // bata instanced (band dinding & pelana mengikuti tahapan; tertutup saat plester)
   542→    const tinggiW = model.config.tinggi;
   543→    const bricks = model.bricks.filter(
   544→      (b) => b.wallId === w.id && (b.z < tinggiW + 0.05 ? V.dinding : V.pelana),
   545→    );
   546→    if (!V.plester && bricks.length > 0) {
   547→      const geo = new THREE.BoxGeometry(1, 1, 1);
   548→      const im = new THREE.InstancedMesh(geo, this.matBrick, bricks.length);
   549→      im.castShadow = true;
   550→      im.receiveShadow = true;
   551→      const m4 = new THREE.Matrix4();
   552→      const q = new THREE.Quaternion();
   553→      const col = new THREE.Color();
   554→      const base = new THREE.Color(model.config.jenis === "hebel" ? 0xeceae3 : 0xa3492f);
   555→      bricks.forEach((b, i) => {
   556→        q.setFromAxisAngle(new THREE.Vector3(0, 1, 0), b.rotY);
   557→        m4.compose(
   558→          new THREE.Vector3(b.x, b.z + b.hgt / 2, b.y),
   559→          q,
   560→          new THREE.Vector3(b.len, b.hgt, t + 0.008),
   561→        );
   562→        im.setMatrixAt(i, m4);
   563→        // variasi warna batu demi batu (deterministik)
   564→        const r = mulberry32(i * 7919 + w.lo * 100)();
   565→        col.copy(base).offsetHSL(0, model.config.jenis === "hebel" ? 0 : 0.01 * (r - 0.5), (r - 0.5) * (model.config.jenis === "hebel" ? 0.05 : 0.08));
   566→        im.setColorAt(i, col);
   567→      });
   568→      group.add(im);
   569→      this.disposables.push(geo);
   570→    }
   571→
   572→    // plesteran + acian / cat: lapisan tipis pada kedua muka dinding (termasuk pelana)
   573→    if (V.plester) {
   574→      const pp = 0.013; // tebal plester tiap muka
   575→      const extSide = w.outer ? (w.c < 3 ? -1 : 1) : 0; // muka luar (searah keluar denah)
   576→      for (const side of [-1, 1]) {
   577→        const isExt = w.outer && side === extSide;
   578→        const mat = V.cat ? (isExt ? this.matCatExt : this.matCatInt) : this.matPlester;
   579→        for (const s of segs) {
   580→          if (s.hi - s.lo < 0.01 || s.z1 - s.z0 < 0.01) continue;
   581→          if (w.axis === "h")
   582→            this.addBox(group, (s.lo + s.hi) / 2, (s.z0 + s.z1) / 2, w.c + side * (t / 2 + pp / 2), s.hi - s.lo, s.z1 - s.z0, pp, mat);
   583→          else
   584→            this.addBox(group, w.c + side * (t / 2 + pp / 2), (s.z0 + s.z1) / 2, (s.lo + s.hi) / 2, pp, s.z1 - s.z0, s.hi - s.lo, mat);
   585→        }
   586→        if (w.gable && V.pelana) {
   587→          const baseZ = w.gable.baseZ;
   588→          const shape = new THREE.Shape();
   589→          shape.moveTo(w.lo, baseZ);
   590→          shape.lineTo(w.hi, baseZ);
   591→          shape.lineTo(w.gable.apexAt, baseZ + w.gable.apexAdd);
   592→          shape.closePath();
   593→          const geo = new THREE.ExtrudeGeometry(shape, { depth: pp, bevelEnabled: false });
   594→          const mesh = new THREE.Mesh(geo, mat);
   595→          mesh.castShadow = true;
   596→          mesh.receiveShadow = true;
   597→          mesh.position.set(0, 0, w.c + side * w.t / 2 + (side > 0 ? 0 : -pp));
   598→          group.add(mesh);
   599→          this.disposables.push(geo);
   600→        }
   601→      }
   602→    }
   603→
   604→    // kusen berprofil, pintu, jendela & lubang angin
   605→    if (V.kusen) {
   606→      for (const o of w.openings) this.buildKusen(group, w, o);
   607→    }
   608→  }
   609→
   610→  /**
   611→   * Kusen berprofil (tiang + ambang + lis dalam membentuk alur), daun pintu
   612→   * panel 2 susun + handle, jendela kaca + palang aluminium, dan lubang angin
   613→   * permanen (roster beton dengan kisi).
   614→   */
   615→  private buildKusen(group: THREE.Group, w: WallRT, o: OpeningSpec) {
   616→    const t = w.t;
   617→    const wdt = o.to - o.from;
   618→    const midO = (o.from + o.to) / 2;
   619→    const ft = t + 0.03; // lebar kusen menonjol sedikit dari muka dinding
   620→    const jw = 0.06; // lebar tiang & ambang kusen
   621→    const lw = 0.024; // lebar lis profil dalam
   622→    const ld = ft * 0.5; // tebal lis profil
   623→    const hgt = o.top - o.sill;
   624→    const cy = (o.sill + o.top) / 2;
   625→
   626→    // helper: kotak pada bidang bukaan (x = sepanjang dinding, y = elevasi)
   627→    const put = (x: number, y: number, sx: number, sy: number, sz: number, mat: THREE.Material, receive = true) => {
   628→      if (w.axis === "h") this.addBox(group, x, y, w.c, sx, sy, sz, mat, receive);
   629→      else this.addBox(group, w.c, y, x, sz, sy, sx, mat, receive);
   630→    };
   631→
   632→    if (o.kind === "vent") {
   633→      // lubang angin permanen: rongga gelap + bingkai & kisi roster beton
   634→      const fr = 0.045;
   635→      put(midO, cy, wdt, hgt, t + 0.012, this.matVent);
   636→      put(midO, o.top - fr / 2, wdt, fr, t + 0.03, this.matVentFrame);
   637→      put(midO, o.sill + fr / 2, wdt, fr, t + 0.03, this.matVentFrame);
   638→      put(o.from + fr / 2, cy, fr, hgt - 2 * fr, t + 0.03, this.matVentFrame);
   639→      put(o.to - fr / 2, cy, fr, hgt - 2 * fr, t + 0.03, this.matVentFrame);
   640→      const inner = wdt - 2 * fr;
   641→      const n = Math.max(4, Math.round(inner / 0.07));
   642→      for (let i = 0; i < n; i++) {
   643→        const x = o.from + fr + (inner * (i + 0.5)) / n;
   644→        put(x, cy, 0.026, hgt - 2 * fr, t + 0.04, this.matVentFrame);
   645→      }
   646→      return;
   647→    }
   648→
   649→    // kotak utama kusen: tiang kiri-kanan, ambang atas, ambang bawah (jendela)
   650→    put(o.from + jw / 2, cy, jw, hgt, ft, this.matKusen);
   651→    put(o.to - jw / 2, cy, jw, hgt, ft, this.matKusen);
   652→    put(midO, o.top - jw / 2, wdt, jw, ft, this.matKusen);
   653→    if (o.sill > 0) put(midO, o.sill + jw / 2, wdt, jw, ft, this.matKusen);
   654→    // lis profil dalam (rebate) menonjol ke bukaan — kesan kusen berprofil
   655→    const ih = hgt - 2 * jw - 0.02;
   656→    put(o.from + jw + lw / 2, cy, lw, ih, ld, this.matKusen);
   657→    put(o.to - jw - lw / 2, cy, lw, ih, ld, this.matKusen);
   658→    put(midO, o.top - jw - lw / 2, wdt - 2 * jw, lw, ld, this.matKusen);
   659→    if (o.sill > 0) put(midO, o.sill + jw + lw / 2, wdt - 2 * jw, lw, ld, this.matKusen);
   660→
   661→    if (o.kind === "pintu") {
   662→      // daun pintu panel + lis 2 susun + handle kedua muka
   663→      const dw = wdt - 2 * jw - 0.015;
   664→      const dh = hgt - 2 * jw - 0.02;
   665→      const dy = o.sill + jw + dh / 2 + 0.008;
   666→      put(midO, dy, dw, dh, 0.045, this.matPintu);
   667→      put(midO, dy + dh * 0.18, dw * 0.6, dh * 0.3, 0.052, this.matPintuGelap);
   668→      put(midO, dy - dh * 0.22, dw * 0.6, dh * 0.42, 0.052, this.matPintuGelap);
   669→      put(midO + dw / 2 - 0.09, o.sill + 1.02, 0.03, 0.14, ft + 0.014, this.matBaja);
   670→    } else {
   671→      // jendela: kaca + palang tengah + batang daun (profil aluminium)
   672→      const gw = wdt - 2 * jw - 0.03;
   673→      const gh = hgt - 2 * jw - 0.03;
   674→      put(midO, cy, gw, gh, 0.012, this.matKaca, false);
   675→      put(midO, cy, gw, 0.035, 0.03, this.matBaja);
   676→      const nLeaf = Math.max(2, Math.round(gw / 0.55));
   677→      for (let li = 1; li < nLeaf; li++) {
   678→        const lx = o.from + jw + (gw * li) / nLeaf;
   679→        put(lx, cy, 0.03, gh, 0.035, this.matBaja);
   680→      }
   681→    }
   682→  }

   684→  /** Pondasi cakar ayam (sistem Sedyatmo): pelat beton 6x6 m + buis beton vertikal. */
   685→  private buildCakarAyam(c: CakarAyamRT) {
   686→    // pelat bertulang: puncak pelat tepat di bawah sloof (-0,2 m), buram agar solid terbaca
   687→    this.addBox(this.houseRoot, 3, -0.2 - CAKAR.slabT / 2, 3, 6, CAKAR.slabT, 6, this.matBeton, true);
   688→    // buis beton menancap ke tanah di tiap titik kolom + grid tengah;
   689→    // ujung atas buis dibuat rata muka pelat agar pola grid cakar terbaca dari atas
   690→    const L = CAKAR.buisL + CAKAR.slabT;
   691→    for (const p of c.buisPts) {
   692→      const geo = new THREE.CylinderGeometry(CAKAR.buisDia / 2, CAKAR.buisDia / 2, L, 18);
   693→      const m = new THREE.Mesh(geo, this.matBuis);
   694→      m.position.set(p.x, -0.2 + 0.02 - L / 2, p.y); // ujung atas sedikit menonjol 2 cm di muka pelat
   695→      m.receiveShadow = true;
   696→      this.houseRoot.add(m);
   697→      this.disposables.push(geo);
   698→    }
   699→  }
   700→
   701→  /**
   702→   * Atap pelana dengan ridge memanjang sepanjang Z (di x = 3).
   703→   * Bidang atap menurun ke dinding kiri (x = 0) dan kanan (x = 6);
   704→   * dinding ujung pelana adalah dinding depan (z = 6) dan belakang (z = 0).
   705→   * Tahapan: rangka atap (kuda-kuda + web + reng + lisplank) -> genteng + nok.
   706→   */
   707→  private buildRoof(model: HouseModel, V: StageVis) {
   708→    const tinggi = model.config.tinggi;
   709→    const apex = model.roof.apexAdd;
   710→    const pitch = model.roof.pitch; // tan(kemiringan)
   711→    const ov = model.roof.overhang;
   712→    const rBase = tinggi + RB_H; // puncak ring balk = dasar dinding pelana / eaves
   713→    const yRidge = model.roof.yRidge; // rBase + apex
   714→    const ang = Math.atan(pitch);
   715→    const zLen = 6 + 2 * ov;
   716→
   717→    // tinggi bidang atap pada jarak horizontal hx dari ridge
   718→    const planeY = (hx: number) => yRidge - hx * pitch;
   719→
   720→    // plafon mengikuti area bersih
   721→    if (V.plafon) {
   722→      const cw = 6 - 2 * (model.config.tLuar / 1000);
   723→      this.addBox(this.roofGroup, 3, tinggi + 0.02, 3, cw, 0.03, cw, this.matPlafon);
   724→    }
   725→
   726→    // ---------- rangka baja ringan (kuda-kuda + web pengaku + reng + lisplank) ----------
   727→    if (V.rangka) {
   728→      // kuda-kuda: rangka C75 pada jarak 1,2 m, tapak di atas ring balk
   729→      const kuda = model.roof.kudaKuda;
   730→      for (let i = 0; i < kuda; i++) {
   731→        const z = 0.3 + i * ((6 - 0.6) / (kuda - 1));
   732→        // ikatan bawah (bottom chord) sepanjang dinding kiri-kanan
   733→        this.addBox(this.trussGroup, 3, rBase + 0.05, z, 5.6, 0.075, 0.06, this.matBaja);
   734→        // dua kasau (top chord) naik ke ridge
   735→        const dx = 2.75;
   736→        const dy = apex - 0.12;
   737→        const L = Math.hypot(dx, dy);
   738→        const angR = Math.atan2(dy, dx);
   739→        const cyR = (rBase + 0.09 + yRidge - 0.03) / 2;
   740→        // kiri: ujung +X (ridge) harus NAIK -> rotasi positif
   741→        this.addTiltedBox(this.trussGroup, 3 - dx / 2, cyR, z, L, 0.075, 0.06, angR, this.matBaja);
   742→        // kanan: ujung -X (ridge) naik -> rotasi negatif
   743→        this.addTiltedBox(this.trussGroup, 3 + dx / 2, cyR, z, L, 0.075, 0.06, -angR, this.matBaja);
   744→        // tiang tengah (king post)
   745→        if (apex > 0.5) {
   746→          this.addBox(this.trussGroup, 3, rBase + 0.09 + (apex - 0.14) / 2, z, 0.05, apex - 0.14, 0.05, this.matBaja);
   747→        }
   748→        // web diagonal pengaku: bottom chord -> kasau -> tiang tengah
   749→        const kasaY = planeY(0.85) + 0.03;
   750→        const postMid = rBase + 0.09 + (apex - 0.14) * 0.55;
   751→        for (const sgn of [-1, 1]) {
   752→          this.addStrut(this.trussGroup, 3 + sgn * 1.5, rBase + 0.087, 3 + sgn * 0.85, kasaY, z, this.matBaja);
   753→          this.addStrut(this.trussGroup, 3 + sgn * 0.85, kasaY, 3, postMid, z, this.matBaja);
   754→        }
   755→      }
   756→
   757→      // reng: melintang sepanjang Z di atas kasau, jarak ~40 cm sepanjang bidang
   758→      for (const s of [-1, 1]) {
   759→        for (let hx = 0.35; hx <= 3 + ov - 0.05; hx += 0.4) {
   760→          this.addTiltedBox(
   761→            this.trussGroup,
   762→            3 + s * hx,
   763→            planeY(hx) + 0.075,
   764→            3,
   765→            0.05,
   766→            0.05,
   767→            zLen,
   768→            s > 0 ? -ang : ang,
   769→            this.matBaja,
   770→          );
   771→        }
   772→      }
   773→
   774→      // lisplank pada kedua ujung eaves (terpasang di rangka sebelum genteng)
   775→      for (const s of [-1, 1]) {
   776→        const x = 3 + s * (3 + ov) - s * 0.02;
   777→        this.addBox(this.trussGroup, x, planeY(3 + ov) - 0.1, 3, 0.035, 0.2, zLen, this.matLisplank);
   778→      }
   779→    }
   780→
   781→    // ---------- genteng + nok ----------
   782→    if (V.genteng) {
   783→      // genteng beton flat: baris sepanjang Z, tumpukan tiap ~40 cm sepanjang bidang
   784→      const dxTile = 0.4 * Math.cos(ang);
   785→      const tileL = dxTile * 1.04;
   786→      const nRows = Math.max(2, Math.ceil((3 + ov) / dxTile));
   787→      const zCount = Math.max(2, Math.ceil(zLen / 0.26));
   788→      for (const s of [-1, 1]) {
   789→        const geo = new THREE.BoxGeometry(tileL, 0.02, 0.24);
   790→        const im = new THREE.InstancedMesh(geo, this.matGenteng, nRows * zCount);
   791→        im.castShadow = true;
   792→        im.receiveShadow = true;
   793→        const m4 = new THREE.Matrix4();
   794→        const q = new THREE.Quaternion();
   795→        const axis = new THREE.Vector3(0, 0, 1);
   796→        const col = new THREE.Color();
   797→        const base = new THREE.Color(0x96524a);
   798→        let idx = 0;
   799→        for (let i = 0; i < nRows; i++) {
   800→          const hx = (i + 0.5) * dxTile;

   809→            im.setMatrixAt(idx, m4);

   832→  private rebuildDims() {
   833→    this.clearGroup(this.dimsGroup);
   834→    const m = this.lastModel;

   837→    const t = m.config.tinggi;

   845→
   846→    // dimensi tinggi total di sudut kanan-depan; label level zigzag agar tidak menumpuk

   864→  private dimLine(a: THREE.Vector3, b: THREE.Vector3, label: string) {
   865→    const g = new THREE.BufferGeometry().setFromPoints([a, b]);
   866→    const line = new THREE.Line(g, this.matDim);
   867→    line.renderOrder = 999;
   868→    this.dimsGroup.add(line);
   869→    this.disposables.push(g);
   870→    const dir = new THREE.Vector3().subVectors(b, a).normalize();
   871→    const perp = Math.abs(dir.y) > 0.9 ? new THREE.Vector3(1, 0, 0) : new THREE.Vector3(0, 1, 0).cross(dir).normalize();
   872→    this.dimTick(a, perp);
   873→    this.dimTick(b, perp);
   874→    const sp = this.makeDimSprite(label);
   875→    sp.position.copy(new THREE.Vector3().addVectors(a, b).multiplyScalar(0.5)).addScaledVector(perp, 0.3);
   876→    this.dimsGroup.add(sp);
   877→  }

  1010→  return Math.max(0, Math.round(v || 0));

  1013→function mulberry32(a: number) {
  1014→  return function () {
  1015→    a |= 0;
  1016→    a = (a + 0x6d2b79f5) | 0;
  1017→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
  1018→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
  1019→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  1020→  };
  1021→}
