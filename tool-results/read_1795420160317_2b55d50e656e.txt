   288→  private onDown = (e: PointerEvent) => {
   289→    this.canvas.setPointerCapture(e.pointerId);
   290→    this.pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
   291→    this.panMode = e.button === 1 || e.button === 2 || e.shiftKey;
   292→    this.downPos = { x: e.clientX, y: e.clientY, t: performance.now(), button: e.button };
   293→    if (this.pointers.size === 2) {
   294→      const pts = [...this.pointers.values()];
   295→      this.pinchDist = Math.hypot(pts[0].x - pts[1].x, pts[0].y - pts[1].y);
   296→    }
   297→  };
   298→
   299→  private onMove = (e: PointerEvent) => {
   300→    const prev = this.pointers.get(e.pointerId);
   301→    if (!prev) {
   302→      // gerakan kursor tanpa tombol tertekan: cek bagian di bawah kursor (tooltip)
   303→      this.hoverAt(e);
   304→      return;
   305→    }
   306→    const dx = e.clientX - prev.x;
   307→    const dy = e.clientY - prev.y;
   308→    this.pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
   309→
   310→    if (this.pointers.size === 2) {
   311→      const pts = [...this.pointers.values()];
   312→      const d = Math.hypot(pts[0].x - pts[1].x, pts[0].y - pts[1].y);
   313→      if (this.pinchDist > 0) this.sph.radius = THREE.MathUtils.clamp(this.sph.radius * (this.pinchDist / d), 3, 45);
   314→      this.pinchDist = d;
   315→      return;
   316→    }
   317→    if (this.panMode) this.pan(dx, dy);
   318→    else {
   319→      this.sph.theta -= dx * 0.0055;
   320→      this.sph.phi = THREE.MathUtils.clamp(this.sph.phi - dy * 0.0045, 0.06, Math.PI * 0.495);
   321→    }
   322→  };
   323→
   324→  private onUp = (e: PointerEvent) => {
   325→    const dp = this.downPos;
   326→    this.pointers.delete(e.pointerId);
   327→    if (this.pointers.size < 2) this.pinchDist = 0;
   328→    this.downPos = null;
   329→    // klik tunggal tanpa seret/pan = pilih bagian bangunan (raycast);
   330→    // klik pada ruang kosong = batalkan seleksi
   331→    if (dp && dp.button === 0 && e.button === 0 && !this.panMode) {
   332→      const dist = Math.hypot(e.clientX - dp.x, e.clientY - dp.y);
   333→      if (dist < 6 && performance.now() - dp.t < 600) this.pickAt(e.clientX, e.clientY);
   334→    }
   335→  };
   336→
   337→  private onWheel = (e: WheelEvent) => {
   338→    e.preventDefault();
   339→    this.sph.radius = THREE.MathUtils.clamp(this.sph.radius * (1 + e.deltaY * 0.0011), 3, 45);
   340→  };
   341→
   342→  private pan(dx: number, dy: number) {
   343→    const fwd = new THREE.Vector3().subVectors(this.target, this.camPos()).setY(0).normalize();
   344→    const right = new THREE.Vector3().crossVectors(new THREE.Vector3(0, 1, 0), fwd).normalize();
   345→    const k = this.sph.radius * 0.0012;
   346→    this.target.addScaledVector(right, -dx * k);
   347→    this.target.addScaledVector(fwd, -dy * k);
   348→    this.target.x = THREE.MathUtils.clamp(this.target.x, -6, 12);
   349→    this.target.z = THREE.MathUtils.clamp(this.target.z, -6, 12);
   350→    this.target.y = 1.2;
   351→  }

   354→    if (p === "depan") {
   355→      this.target.set(3, 2.0, 3);

   357→    } else if (p === "belakang") {
   358→      this.target.set(3, 2.0, 3);
   359→      this.sph = { radius: 12.5, theta: Math.PI, phi: 1.3 };
   360→    } else if (p === "samping") {
   361→      this.target.set(3, 2.0, 3);
   362→      this.sph = { radius: 12.5, theta: Math.PI / 2, phi: 1.3 };
   363→    } else if (p === "atas") {
   364→      this.target.set(3, 0, 3);
   365→      this.sph = { radius: 13.5, theta: Math.PI * 0.001, phi: 0.03 };
   366→    } else {
   367→      this.target.set(3, 1.4, 3);
   368→      this.sph = { radius: 13, theta: Math.PI * 0.15, phi: 1.02 };
   369→    }
   370→  }

   394→    this.dimsOn = on;
   395→    this.rebuildDims();

   400→  /** Daftarkan callback dari React: pilih bagian (klik) & tooltip hover. */
   401→  setPickHandlers(
   402→    h: {
   403→      onPick?: (part: string | null) => void;
   404→      onHover?: (part: string | null, x: number, y: number) => void;
   405→    },
   406→  ) {
   407→    this.pickHandlers = h;
   408→  }

   410→  /** Set bagian terpilih (highlight emissive di 3D); null = tanpa seleksi. */
   411→  setSelection(part: string | null) {
   412→    if (this.selectedPart === part) return;
   413→    this.selectedPart = part;
   414→    this.restoreHighlight();
   415→    this.applyHighlight();
   416→  }
   417→
   418→  private restoreHighlight() {
   419→    for (const h of this.hiObjs) h.obj.material = h.mat;
   420→    this.hiObjs.length = 0;
   421→  }
   422→
   423→  /** Highlight seluruh mesh milik bagian terpilih (klon material + emissive). */
   424→  private applyHighlight() {
   425→    const part = this.selectedPart;
   426→    if (!part) return;
   427→    for (const root of this.pickables) {
   428→      if (root.userData.pick !== part) continue;
   429→      root.traverse((c) => {
   430→        const mesh = c as THREE.Mesh;
   431→        if (!mesh.isMesh || !mesh.material) return;
   432→        const orig = mesh.material as THREE.Material;
   433→        let hi = this.hiMats.get(orig);
   434→        if (!hi) {
   435→          hi = orig.clone();
   436→          const std = hi as THREE.MeshStandardMaterial;
   437→          if (std.isMeshStandardMaterial) {
   438→            std.emissive = new THREE.Color(0xd97a2b);
   439→            std.emissiveIntensity = 0.45;
   440→          }
   441→          hi.clippingPlanes = [this.clipPlane];
   442→          (hi as THREE.MeshStandardMaterial).wireframe = (orig as THREE.MeshStandardMaterial).wireframe;
   443→          this.hiMats.set(orig, hi);
   444→        }
   445→        this.hiObjs.push({ obj: mesh, mat: orig });
   446→        mesh.material = hi;
   447→      });
   448→    }
   449→  }

   451→  /** Raycast dari koordinat layar; id bagian via walk-up ke objek/grup bertag. */
   452→  private raycastPart(cx: number, cy: number): string | null {
   453→    const rect = this.canvas.getBoundingClientRect();
   454→    if (rect.width < 2 || rect.height < 2) return null;
   455→    const ndc = new THREE.Vector2(
   456→      ((cx - rect.left) / rect.width) * 2 - 1,
   457→      -((cy - rect.top) / rect.height) * 2 + 1,
   458→    );
   459→    const cam: THREE.Camera = this.useOrtho ? this.cameraOrtho : this.camera;
   460→    this.raycaster.setFromCamera(ndc, cam);
   461→    const hits = this.raycaster.intersectObjects(this.pickables, true);
   462→    for (const h of hits) {
   463→      let o: THREE.Object3D | null = h.object;
   464→      while (o) {
   465→        if (o.userData?.pick) return o.userData.pick as string;
   466→        o = o.parent;
   467→      }
   468→    }
   469→    return null;
   470→  }
   471→
   472→  /** Klik kanvas: pilih bagian & umpankan id-nya ke React. */
   473→  private pickAt(cx: number, cy: number) {
   474→    const part = this.raycastPart(cx, cy);
   475→    if (part !== this.selectedPart) this.setSelection(part);
   476→    this.pickHandlers.onPick?.(part);
   477→  }
   478→
   479→  /** Hover tanpa tekan: tooltip nama bagian + kursor pointer (dithrottle 70 ms). */
   480→  private hoverAt(e: PointerEvent) {
   481→    const now = performance.now();
   482→    if (now - this.hoverThrottle < 70) return;
   483→    this.hoverThrottle = now;
   484→    const part = this.raycastPart(e.clientX, e.clientY);
   485→    this.canvas.style.cursor = part ? "pointer" : "";
   486→    if (part !== this.lastHover) {
   487→      this.lastHover = part;
   488→      const rect = this.canvas.getBoundingClientRect();
   489→      this.pickHandlers.onHover?.(part, e.clientX - rect.left, e.clientY - rect.top);
   490→    }
   491→  }
   492→
   493→  private onLeave = () => {
   494→    this.lastHover = null;
   495→    this.pickHandlers.onHover?.(null, 0, 0);
   496→    this.canvas.style.cursor = "";
   497→  };
   498→
   499→  /** Tandai objek (mesh / grup) sebagai bagian yang bisa diklik (raycast). */
   500→  private tag(obj: THREE.Object3D, part: string) {
   501→    obj.userData.pick = part;
   502→    this.pickables.push(obj);
   503→    return obj;
   504→  }
   505→
   506→  private camPos(): THREE.Vector3 {
   507→    const { radius, theta, phi } = this.sph;
   508→    return new THREE.Vector3(
   509→      this.target.x + radius * Math.sin(phi) * Math.sin(theta),
   510→      this.target.y + radius * Math.cos(phi),
   511→      this.target.z + radius * Math.sin(phi) * Math.cos(theta),
   512→    );
   513→  }
   514→
   515→  private loop = () => {
   516→    this.raf = requestAnimationFrame(this.loop);
   517→    const cam: THREE.PerspectiveCamera | THREE.OrthographicCamera = this.useOrtho ? this.cameraOrtho : this.camera;
   518→    cam.position.copy(this.camPos());
   519→    cam.lookAt(this.target);
   520→    if (this.useOrtho) this.syncOrtho();
   521→    this.renderer.render(this.scene, cam);
   522→  };
   523→
   524→  private syncOrtho() {
   525→    const el = this.canvas.parentElement ?? this.canvas;
   526→    const w = Math.max(1, el.clientWidth);
   527→    const h = Math.max(1, el.clientHeight);
   528→    const halfH = this.sph.radius * Math.tan((this.camera.fov * Math.PI) / 360);
   529→    const halfW = halfH * (w / h);
   530→    const o = this.cameraOrtho;
   531→    o.left = -halfW;
   532→    o.right = halfW;
   533→    o.top = halfH;
   534→    o.bottom = -halfH;
   535→    o.updateProjectionMatrix();
   536→  }
   537→
   538→  private resize() {
   539→    const el = this.canvas.parentElement ?? this.canvas;
   540→    const w = Math.max(1, el.clientWidth);
   541→    const h = Math.max(1, el.clientHeight);
   542→    this.renderer.setSize(w, h, false);
   543→    this.camera.aspect = w / h;
   544→    this.camera.updateProjectionMatrix();
   545→    this.syncOrtho();
   546→  }

   685→    // anotasi dimensi 3D mengikuti model terakhir
   686→    this.rebuildDims();
   687→
   688→    // pasang ulang highlight bagian terpilih pada mesh yang baru dibangun

   690→  }

   692→  private buildWall(w: WallRT, model: HouseModel, V: StageVis, cakar: boolean, hid: (p: string) => boolean) {
   693→    const group = new THREE.Group();
   694→    group.userData.explodeDir = w.explodeDir;
   695→    this.houseRoot.add(group);
   696→
   697→    const len = w.hi - w.lo;
   698→    const t = w.t;
   699→    const mid = (w.lo + w.hi) / 2;
   700→    const bw = Math.max(0.15, t + 0.01); // lebar sloof / ring balk (15 cm standar)
   701→
   702→    // pondasi batu kali: strip 40 x 60 cm dari -0,2 s/d -0,8 m, disusun batu demi batu
   703→    // (segmen dengan tinggi/lebar bervariasi agar terbaca sebagai batu kali, bukan beton)
   704→    if (!cakar && V.pondasi) {
   705→      const rng = mulberry32(Math.round((w.lo + w.c) * 977));
   706→      let a = w.lo;
   707→      while (a < w.hi - 0.05) {
   708→        const segLen = Math.min(0.3 + rng() * 0.28, w.hi - a);
   709→        const segW = PONDASI_W * (0.86 + rng() * 0.14);
   710→        const segH = PONDASI_D * (0.82 + rng() * 0.18);
   711→        const cy = PONDASI_TOP - segH / 2 + (rng() - 0.5) * 0.03;
   712→        if (w.axis === "h") {
   713→          this.addBox(group, a + segLen / 2, cy, w.c + (rng() - 0.5) * 0.02, segLen, segH, segW, this.matPlinth, true, "pondasi");
   714→        } else {
   715→          this.addBox(group, w.c + (rng() - 0.5) * 0.02, cy, a + segLen / 2, segW, segH, segLen, this.matPlinth, true, "pondasi");
   716→        }
   717→        a += segLen;
   718→      }
   719→    }
   720→    if (V.sloof) {
   721→      if (w.axis === "h") this.addBox(group, mid, PONDASI_TOP + RB_H / 2, w.c, len, RB_H, bw, this.matBeton, true, "sloof");
   722→      else this.addBox(group, w.c, PONDASI_TOP + RB_H / 2, mid, bw, RB_H, len, this.matBeton, true, "sloof");
   723→    }
   724→    if (V.ringbalk) {
   725→      if (w.axis === "h") this.addBox(group, mid, w.h + RB_H / 2, w.c, len, RB_H, bw, this.matBeton, true, "ringbalk");
   726→      else this.addBox(group, w.c, w.h + RB_H / 2, mid, bw, RB_H, len, this.matBeton, true, "ringbalk");
   727→    }
   728→
   729→    // mortar/nat slab: kotak-kotak menghindari bukaan (hanya sampai tinggi dinding datar)
   730→    const segs: { lo: number; hi: number; z0: number; z1: number }[] = [];
   731→    const ops = [...w.openings].sort((a, b) => a.from - b.from);
   732→    let cursor = w.lo;
   733→    for (const o of ops) {
   734→      const clo = Math.max(o.from, w.lo);
   735→      const chi = Math.min(o.to, w.hi);
   736→      if (clo > cursor) segs.push({ lo: cursor, hi: clo, z0: 0, z1: w.h });
   737→      if (o.sill > 0) segs.push({ lo: clo, hi: chi, z0: 0, z1: o.sill });
   738→      if (o.top < w.h) segs.push({ lo: clo, hi: chi, z0: o.top, z1: w.h });
   739→      cursor = Math.max(cursor, chi);
   740→    }
   741→    if (cursor < w.hi) segs.push({ lo: cursor, hi: w.hi, z0: 0, z1: w.h });
   742→
   743→    // mortar/nat terlihat di sela bata — tertutup begitu plester dipasang
   744→    if (V.dinding && !V.plester) {
   745→      for (const s of segs) {
   746→        if (s.hi - s.lo < 0.01 || s.z1 - s.z0 < 0.01) continue;
   747→        if (w.axis === "h") this.addBox(group, (s.lo + s.hi) / 2, (s.z0 + s.z1) / 2, w.c, s.hi - s.lo, s.z1 - s.z0, t, this.matMortar, true, "dinding");
   748→        else this.addBox(group, w.c, (s.z0 + s.z1) / 2, (s.lo + s.hi) / 2, t, s.z1 - s.z0, s.hi - s.lo, this.matMortar, true, "dinding");
   749→      }
   750→    }
   751→
   752→    // segitiga mortar di ujung atap pelana (dinding depan/belakang), di atas ring balk
   753→    if (w.gable && V.pelana && !V.plester) {
   754→      const baseZ = w.gable.baseZ;
   755→      const shape = new THREE.Shape();
   756→      shape.moveTo(w.lo, baseZ);
   757→      shape.lineTo(w.hi, baseZ);
   758→      shape.lineTo(w.gable.apexAt, baseZ + w.gable.apexAdd);
   759→      shape.closePath();
   760→      const geo = new THREE.ExtrudeGeometry(shape, { depth: t, bevelEnabled: false });
   761→      const mesh = new THREE.Mesh(geo, this.matMortar);
   762→      mesh.castShadow = true;
   763→      mesh.receiveShadow = true;
   764→      mesh.position.set(0, 0, w.c - t / 2);
   765→      this.tag(mesh, "dinding");
   766→      group.add(mesh);
   767→      this.dyn.push(geo);
   768→    }
   769→
   770→    // bata instanced (band dinding & pelana mengikuti tahapan; tertutup saat plester)
   771→    const tinggiW = model.config.tinggi;
   772→    const bricks = model.bricks.filter(
   773→      (b) => b.wallId === w.id && (b.z < tinggiW + 0.05 ? V.dinding : V.pelana),
   774→    );
   775→    if (!V.plester && bricks.length > 0) {
   776→      const geo = new THREE.BoxGeometry(1, 1, 1);
   777→      const im = new THREE.InstancedMesh(geo, this.matBrick, bricks.length);
   778→      im.castShadow = true;
   779→      im.receiveShadow = true;
   780→      this.tag(im, "dinding");
   781→      const m4 = new THREE.Matrix4();
   782→      const q = new THREE.Quaternion();
   783→      const col = new THREE.Color();
   784→      const base = new THREE.Color(model.config.jenis === "hebel" ? 0xeceae3 : 0xa3492f);
   785→      bricks.forEach((b, i) => {
   786→        q.setFromAxisAngle(new THREE.Vector3(0, 1, 0), b.rotY);
   787→        m4.compose(
   788→          new THREE.Vector3(b.x, b.z + b.hgt / 2, b.y),
   789→          q,
   790→          new THREE.Vector3(b.len, b.hgt, t + 0.008),
   791→        );
   792→        im.setMatrixAt(i, m4);
   793→        // variasi warna batu demi batu (deterministik)
   794→        const r = mulberry32(i * 7919 + w.lo * 100)();
   795→        col.copy(base).offsetHSL(0, model.config.jenis === "hebel" ? 0 : 0.01 * (r - 0.5), (r - 0.5) * (model.config.jenis === "hebel" ? 0.05 : 0.08));
   796→        im.setColorAt(i, col);
   797→      });
   798→      group.add(im);
   799→      this.dyn.push(geo);
   800→    }
   801→
   802→    // plesteran + acian / cat: lapisan tipis pada kedua muka dinding (termasuk pelana)
   803→    if (V.plester) {
   804→      const pp = 0.013; // tebal plester tiap muka
   805→      const extSide = w.outer ? (w.c < 3 ? -1 : 1) : 0; // muka luar (searah keluar denah)
   806→      for (const side of [-1, 1]) {
   807→        const isExt = w.outer && side === extSide;
   808→        const mat = V.cat ? (isExt ? this.matCatExt : this.matCatInt) : this.matPlester;
   809→        for (const s of segs) {
   810→          if (s.hi - s.lo < 0.01 || s.z1 - s.z0 < 0.01) continue;
   811→          if (w.axis === "h")
   812→            this.addBox(group, (s.lo + s.hi) / 2, (s.z0 + s.z1) / 2, w.c + side * (t / 2 + pp / 2), s.hi - s.lo, s.z1 - s.z0, pp, mat, true, "plester");
   813→          else
   814→            this.addBox(group, w.c + side * (t / 2 + pp / 2), (s.z0 + s.z1) / 2, (s.lo + s.hi) / 2, pp, s.z1 - s.z0, s.hi - s.lo, mat, true, "plester");
   815→        }
   816→        if (w.gable && V.pelana) {
   817→          const baseZ = w.gable.baseZ;
   818→          const shape = new THREE.Shape();
   819→          shape.moveTo(w.lo, baseZ);
   820→          shape.lineTo(w.hi, baseZ);
   821→          shape.lineTo(w.gable.apexAt, baseZ + w.gable.apexAdd);
   822→          shape.closePath();
   823→          const geo = new THREE.ExtrudeGeometry(shape, { depth: pp, bevelEnabled: false });
   824→          const mesh = new THREE.Mesh(geo, mat);
   825→          mesh.castShadow = true;
   826→          mesh.receiveShadow = true;
   827→          mesh.position.set(0, 0, w.c + side * w.t / 2 + (side > 0 ? 0 : -pp));
   828→          this.tag(mesh, "plester");
   829→          group.add(mesh);
   830→          this.dyn.push(geo);
   831→        }
   832→      }
   833→    }
   834→
   835→    // kusen berprofil, pintu, jendela & lubang angin
   836→    if (V.kusen) {
   837→      for (const o of w.openings) {
   838→        if (hid(o.kind)) continue; // pintu / jendela / roster bisa disembunyikan per jenis
   839→        this.buildKusen(group, w, o);
   840→      }
   841→    }
   842→  }
   843→
   844→  /**
   845→   * Kusen berprofil (tiang + ambang + lis dalam membentuk alur), daun pintu
   846→   * panel 2 susun + handle, jendela kaca + palang aluminium, dan lubang angin
   847→   * permanen (roster beton dengan kisi).
   848→   */
   849→  private buildKusen(group: THREE.Group, w: WallRT, o: OpeningSpec) {
   850→    const t = w.t;
   851→    const wdt = o.to - o.from;
   852→    const midO = (o.from + o.to) / 2;
   853→    const ft = t + 0.03; // lebar kusen menonjol sedikit dari muka dinding
   854→    const jw = 0.06; // lebar tiang & ambang kusen
   855→    const lw = 0.024; // lebar lis profil dalam
   856→    const ld = ft * 0.5; // tebal lis profil
   857→    const hgt = o.top - o.sill;
   858→    const cy = (o.sill + o.top) / 2;
   859→
   860→    // helper: kotak pada bidang bukaan (x = sepanjang dinding, y = elevasi);
   861→    // tiap mesh kusen di-tag sebagai bagian sesuai jenis bukaannya (pintu/jendela/vent)
   862→    const put = (x: number, y: number, sx: number, sy: number, sz: number, mat: THREE.Material, receive = true) => {
   863→      if (w.axis === "h") this.addBox(group, x, y, w.c, sx, sy, sz, mat, receive, o.kind);
   864→      else this.addBox(group, w.c, y, x, sz, sy, sx, mat, receive, o.kind);
   865→    };
   866→
   867→    if (o.kind === "vent") {
   868→      // lubang angin permanen: rongga gelap + bingkai & kisi roster beton
   869→      const fr = 0.045;
   870→      put(midO, cy, wdt, hgt, t + 0.012, this.matVent);
   871→      put(midO, o.top - fr / 2, wdt, fr, t + 0.03, this.matVentFrame);
   872→      put(midO, o.sill + fr / 2, wdt, fr, t + 0.03, this.matVentFrame);
   873→      put(o.from + fr / 2, cy, fr, hgt - 2 * fr, t + 0.03, this.matVentFrame);
   874→      put(o.to - fr / 2, cy, fr, hgt - 2 * fr, t + 0.03, this.matVentFrame);
   875→      const inner = wdt - 2 * fr;
   876→      const n = Math.max(4, Math.round(inner / 0.07));
   877→      for (let i = 0; i < n; i++) {
   878→        const x = o.from + fr + (inner * (i + 0.5)) / n;
   879→        put(x, cy, 0.026, hgt - 2 * fr, t + 0.04, this.matVentFrame);
   880→      }
   881→      return;
   882→    }
   883→
   884→    // kotak utama kusen: tiang kiri-kanan, ambang atas, ambang bawah (jendela)
   885→    put(o.from + jw / 2, cy, jw, hgt, ft, this.matKusen);
   886→    put(o.to - jw / 2, cy, jw, hgt, ft, this.matKusen);
   887→    put(midO, o.top - jw / 2, wdt, jw, ft, this.matKusen);
   888→    if (o.sill > 0) put(midO, o.sill + jw / 2, wdt, jw, ft, this.matKusen);
   889→    // lis profil dalam (rebate) menonjol ke bukaan — kesan kusen berprofil
   890→    const ih = hgt - 2 * jw - 0.02;
   891→    put(o.from + jw + lw / 2, cy, lw, ih, ld, this.matKusen);
   892→    put(o.to - jw - lw / 2, cy, lw, ih, ld, this.matKusen);
   893→    put(midO, o.top - jw - lw / 2, wdt - 2 * jw, lw, ld, this.matKusen);
   894→    if (o.sill > 0) put(midO, o.sill + jw + lw / 2, wdt - 2 * jw, lw, ld, this.matKusen);
   895→
   896→    if (o.kind === "pintu") {
   897→      // daun pintu panel + lis 2 susun + handle kedua muka
   898→      const dw = wdt - 2 * jw - 0.015;
   899→      const dh = hgt - 2 * jw - 0.02;
   900→      const dy = o.sill + jw + dh / 2 + 0.008;
   901→      put(midO, dy, dw, dh, 0.045, this.matPintu);
   902→      put(midO, dy + dh * 0.18, dw * 0.6, dh * 0.3, 0.052, this.matPintuGelap);
   903→      put(midO, dy - dh * 0.22, dw * 0.6, dh * 0.42, 0.052, this.matPintuGelap);
   904→      put(midO + dw / 2 - 0.09, o.sill + 1.02, 0.03, 0.14, ft + 0.014, this.matBaja);
   905→    } else {
   906→      // jendela: kaca + palang tengah + batang daun (profil aluminium)
   907→      const gw = wdt - 2 * jw - 0.03;
   908→      const gh = hgt - 2 * jw - 0.03;
   909→      put(midO, cy, gw, gh, 0.012, this.matKaca, false);
   910→      put(midO, cy, gw, 0.035, 0.03, this.matBaja);
   911→      const nLeaf = Math.max(2, Math.round(gw / 0.55));
   912→      for (let li = 1; li < nLeaf; li++) {
   913→        const lx = o.from + jw + (gw * li) / nLeaf;
   914→        put(lx, cy, 0.03, gh, 0.035, this.matBaja);
   915→      }
   916→    }
   917→  }

   919→  /** Pondasi cakar ayam (sistem Sedyatmo): pelat beton 6x6 m + buis beton vertikal. */
   920→  private buildCakarAyam(c: CakarAyamRT) {
   921→    // pelat bertulang: puncak pelat tepat di bawah sloof (-0,2 m), buram agar solid terbaca
   922→    this.addBox(this.houseRoot, 3, PONDASI_TOP - CAKAR.slabT / 2, 3, 6, CAKAR.slabT, 6, this.matBeton, true, "pondasi");
   923→    // buis beton menancap ke tanah di tiap titik kolom + grid tengah;
   924→    // ujung atas buis dibuat rata muka pelat agar pola grid cakar terbaca dari atas
   925→    const L = CAKAR.buisL + CAKAR.slabT;
   926→    for (const p of c.buisPts) {
   927→      const geo = new THREE.CylinderGeometry(CAKAR.buisDia / 2, CAKAR.buisDia / 2, L, 18);
   928→      const m = new THREE.Mesh(geo, this.matBuis);
   929→      m.position.set(p.x, PONDASI_TOP + 0.02 - L / 2, p.y); // ujung atas sedikit menonjol 2 cm di muka pelat
   930→      m.receiveShadow = true;
   931→      this.tag(m, "pondasi");
   932→      this.houseRoot.add(m);
   933→      this.dyn.push(geo);
   934→    }
   935→  }
   936→
   937→  /**
   938→   * Atap pelana dengan ridge memanjang sepanjang Z (di x = 3).
   939→   * Bidang atap menurun ke dinding kiri (x = 0) dan kanan (x = 6);
   940→   * dinding ujung pelana adalah dinding depan (z = 6) dan belakang (z = 0).
   941→   * Tahapan: rangka atap (kuda-kuda + web + reng + lisplank) -> genteng + nok.
   942→   */
   943→  private buildRoof(model: HouseModel, V: StageVis, hid: (p: string) => boolean) {
   944→    const tinggi = model.config.tinggi;
   945→    const apex = model.roof.apexAdd;
   946→    const pitch = model.roof.pitch; // tan(kemiringan)
   947→    const ov = model.roof.overhang;
   948→    const rBase = tinggi + RB_H; // puncak ring balk = dasar dinding pelana / eaves
   949→    const yRidge = model.roof.yRidge; // rBase + apex
   950→    const ang = Math.atan(pitch);
   951→    const zLen = 6 + 2 * ov;
   952→
   953→    // tinggi bidang atap pada jarak horizontal hx dari ridge
   954→    const planeY = (hx: number) => yRidge - hx * pitch;
   955→
   956→    // plafon mengikuti area bersih
   957→    if (V.plafon) {
   958→      const cw = 6 - 2 * (model.config.tLuar / 1000);
   959→      this.addBox(this.roofGroup, 3, tinggi + 0.02, 3, cw, 0.03, cw, this.matPlafon, true, "plafon");
   960→    }
   961→
   962→    // ---------- rangka baja ringan (kuda-kuda + web pengaku + reng + lisplank) ----------
   963→    if (V.rangka) {
   964→      // kuda-kuda: rangka C75 pada jarak maks 1,2 m, tapak di atas ring balk
   965→      const kuda = model.roof.kudaKuda;
   966→      for (let i = 0; i < kuda; i++) {
   967→        const z = 0.3 + i * ((6 - 0.6) / (kuda - 1));
   968→        // ikatan bawah (bottom chord) sepanjang dinding kiri-kanan
   969→        this.addBox(this.trussGroup, 3, rBase + 0.05, z, 5.6, 0.075, 0.06, this.matBaja);
   970→        // dua kasau (top chord) naik ke ridge; ujung bawah menerus hingga tepi
   971→        // overstek sehingga reng & lisplank di ujung atap tetap bertumpu
   972→        const dx = 3 + ov - 0.06;
   973→        const dy = apex + ov * pitch - 0.12;
   974→        const L = Math.hypot(dx, dy);
   975→        const angR = Math.atan2(dy, dx);
   976→        const cyR = (rBase + 0.09 - ov * pitch + yRidge - 0.03) / 2;
   977→        // kiri: ujung +X (ridge) harus NAIK -> rotasi positif
   978→        this.addTiltedBox(this.trussGroup, 3 - dx / 2, cyR, z, L, 0.075, 0.06, angR, this.matBaja);
   979→        // kanan: ujung -X (ridge) naik -> rotasi negatif
   980→        this.addTiltedBox(this.trussGroup, 3 + dx / 2, cyR, z, L, 0.075, 0.06, -angR, this.matBaja);
   981→        // tiang tengah (king post)
   982→        if (apex > 0.5) {
   983→          this.addBox(this.trussGroup, 3, rBase + 0.09 + (apex - 0.14) / 2, z, 0.05, apex - 0.14, 0.05, this.matBaja);
   984→        }
   985→        // web diagonal pengaku: bottom chord -> kasau -> tiang tengah
   986→        const kasaY = planeY(0.85) + 0.03;
   987→        const postMid = rBase + 0.09 + (apex - 0.14) * 0.55;
   988→        for (const sgn of [-1, 1]) {
   989→          this.addStrut(this.trussGroup, 3 + sgn * 1.5, rBase + 0.087, 3 + sgn * 0.85, kasaY, z, this.matBaja);
   990→          this.addStrut(this.trussGroup, 3 + sgn * 0.85, kasaY, 3, postMid, z, this.matBaja);
   991→        }
   992→      }
   993→
   994→      // reng: melintang sepanjang Z di atas kasau, jarak ~40 cm sepanjang bidang
   995→      // (top reng 1 cm di bawah muka genteng agar tidak z-fighting)
   996→      for (const s of [-1, 1]) {
   997→        for (let hx = 0.35; hx <= 3 + ov - 0.05; hx += 0.4) {
   998→          this.addTiltedBox(
   999→            this.trussGroup,
  1000→            3 + s * hx,
  1001→            planeY(hx) + 0.065,
  1002→            3,
  1003→            0.05,
  1004→            0.05,
  1005→            zLen,
  1006→            s > 0 ? -ang : ang,
  1007→            this.matBaja,
  1008→          );
  1009→        }
  1010→      }
  1011→
  1012→      // lisplank pada kedua ujung eaves (terpasang di rangka sebelum genteng)
  1013→      // (grup terpisah agar bisa disembunyikan / dipilih sebagai bagian sendiri)
  1014→      if (!hid("lisplank")) {
  1015→        for (const s of [-1, 1]) {
  1016→          const x = 3 + s * (3 + ov) - s * 0.02;
  1017→          this.addBox(this.lisplankGroup, x, planeY(3 + ov) - 0.1, 3, 0.035, 0.2, zLen, this.matLisplank, true, "lisplank");
  1018→        }
  1019→
  1020→        // lisplank ujung pelana (menutup overstek pada kedua ujung segitiga atap)
  1021→        const slopeLen = Math.hypot(3 + ov, apex + ov * pitch);
  1022→        const yMidSlope = (yRidge + rBase - ov * pitch) / 2;
  1023→        for (const s of [-1, 1]) {
  1024→          for (const zz of [-ov + 0.017, 6 + ov - 0.017]) {
  1025→            this.addTiltedBox(this.lisplankGroup, 3 + s * (3 + ov) / 2, yMidSlope, zz, slopeLen, 0.2, 0.034, s > 0 ? -ang : ang, this.matLisplank, "lisplank");
  1026→          }
  1027→        }
  1028→      }
  1029→    }
  1030→
  1031→    // ---------- genteng + nok ----------
  1032→    if (V.genteng) {
  1033→      // genteng beton flat: pemasangan rapat (tile eksak baris demi baris sepanjang
  1034→      // bidang sehingga tidak ada celah memperlihatkan reng maupun z-fighting)
  1035→      const dxTile = 0.4 * Math.cos(ang);
  1036→      const nRows = Math.max(2, Math.ceil((3 + ov) / dxTile));
  1037→      const rowSpacing = (3 + ov) / nRows; // panjang genteng = jarak baris (eksak)
  1038→      const zCount = Math.max(2, Math.ceil(zLen / 0.25));
  1039→      const zSpacing = zLen / zCount; // lebar genteng = jarak lajur (eksak)
  1040→      for (const s of [-1, 1]) {
  1041→        const geo = new THREE.BoxGeometry(rowSpacing, 0.02, zSpacing);
  1042→        const im = new THREE.InstancedMesh(geo, this.matGenteng, nRows * zCount);
  1043→        im.castShadow = true;
  1044→        im.receiveShadow = true;
  1045→        const m4 = new THREE.Matrix4();
  1046→        const q = new THREE.Quaternion();
  1047→        const axis = new THREE.Vector3(0, 0, 1);
  1048→        const col = new THREE.Color();
  1049→        const base = new THREE.Color(0x96524a);
  1050→        let idx = 0;
  1051→        for (let i = 0; i < nRows; i++) {
  1052→          const hx = (i + 0.5) * rowSpacing;
  1053→          for (let jj = 0; jj < zCount; jj++) {
  1054→            const z = -ov + (jj + 0.5) * zSpacing;
  1055→            q.setFromAxisAngle(axis, s > 0 ? -ang : ang);
  1056→            m4.compose(
  1057→              new THREE.Vector3(3 + s * hx, planeY(hx) + 0.09, z),
  1058→              q,
  1059→              new THREE.Vector3(1, 1, 1),
  1060→            );
  1061→            im.setMatrixAt(idx, m4);
  1062→            const r = mulberry32(idx * 2654435761 + (s > 0 ? 7 : 13))();
  1063→            col.copy(base).offsetHSL(0.003 * (r - 0.5), 0.025 * (r - 0.5), 0.03 * (r - 0.5));
  1064→            im.setColorAt(idx, col);
  1065→            idx++;
  1066→          }
  1067→        }
  1068→        this.tilesGroup.add(im);
  1069→        this.dyn.push(geo);
  1070→      }
  1071→
  1072→      // nok (ridge cap) menutup pertemuan dua bidang genteng
  1073→      const g = new THREE.BoxGeometry(0.4, 0.09, zLen + 0.06);
  1074→      const nok = new THREE.Mesh(g, this.matRidge);
  1075→      nok.castShadow = true;
  1076→      nok.position.set(3, yRidge + 0.11, 3);
  1077→      this.tilesGroup.add(nok);
  1078→      this.dyn.push(g);
  1079→    }
  1080→  }
  1081→
  1082→  // ------------------------------------------------------- instalasi MEP --
  1083→
  1084→  /**
  1085→   * Instalasi air & listrik (MEP) sebagai overlay sinar-X:
  1086→   * - air bersih PVC AW 1/2" biru (Ø visual = Ø asli 21 mm)
  1087→   * - limbah PVC 3" hijau-teal (Ø 89 mm)
  1088→   * - conduit Ø20 mm kuning (sirkuit lampu, NYM 2x1,5) / merah (stop kontak, 2x2,5)
  1089→   * - titik: lampu (bola kuning emissive), saklar & stop kontak (piring krem),
  1090→   *   panel MCB, meteran air, kran, shower, kloset, floor drain
  1091→   * - septic tank + sumur resapan di belakang rumah (bawah tanah, tembus pandang)
  1092→   */
  1093→  private buildMep(model: HouseModel, air: boolean, listrik: boolean, hid: (p: string) => boolean) {
  1094→    if (!air && !listrik) return;
  1095→    const m = model.mep;
  1096→
  1097→    // Ø visual pipa (meter): 1/2" = 0,021 m; 3" = 0,045 m; conduit 20 mm = 0,015 m
  1098→    const R_BERSIH = 0.021;
  1099→    const R_KOTOR = 0.045;
  1100→    const R_CONDUIT = 0.015;
  1101→
  1102→    // sambungan siku diberi bola agar pertemuan pipa rapat (bukan celah flat-cap)
  1103→    const joint = (p: { x: number; y: number; z: number }, r: number, mat: THREE.Material, pick?: string) =>
  1104→      this.addMepSphere(p.x, p.y, p.z, r, mat, pick);
  1105→
  1106→    // id bagian (parts.ts) untuk tiap jalur — dipakai untuk tag klik & visibilitas
  1107→    const routePart = (r: { jenis: string; sirkuit: "lampu" | "kontak" | null }) =>
  1108→      r.jenis === "air_bersih"
  1109→        ? "pipaAirBersih"
  1110→        : r.jenis === "air_kotor"
  1111→          ? "pipaAirKotor"
  1112→          : r.sirkuit === "kontak"
  1113→            ? "listrikKontak"
  1114→            : "listrikLampu";
  1115→
  1116→    for (const r of m.routes) {
  1117→      const isAir = r.jenis !== "listrik";
  1118→      if (isAir && !air) continue;
  1119→      if (!isAir && !listrik) continue;
  1120→      const part = routePart(r);
  1121→      if (hid(part)) continue;
  1122→      const mat =
  1123→        r.jenis === "air_bersih"
  1124→          ? this.matPipeBersih
  1125→          : r.jenis === "air_kotor"
  1126→            ? this.matPipeKotor
  1127→            : r.sirkuit === "kontak"
  1128→              ? this.matKabelKontak
  1129→              : this.matKabelLampu;
  1130→      const rad = r.jenis === "air_bersih" ? R_BERSIH : r.jenis === "air_kotor" ? R_KOTOR : R_CONDUIT;
  1131→      for (let i = 1; i < r.pts.length; i++) {
  1132→        this.addPipe(r.pts[i - 1], r.pts[i], rad, mat, part);
  1133→      }
  1134→      for (let i = 1; i < r.pts.length - 1; i++) joint(r.pts[i], rad, mat, part);
  1135→    }
  1136→
  1137→    // orientasi piring saklar/stop kontak: tipis mengikuti dinding terdekat
  1138→    const thinZ = (p: { x: number; z: number }) => {
  1139→      let best: "h" | "v" = "h";
  1140→      let bd = Infinity;
  1141→      for (const w of model.walls) {
  1142→        const d = w.axis === "h" ? Math.abs(p.z - w.c) : Math.abs(p.x - w.c);
  1143→        if (d < bd) {
  1144→          bd = d;
  1145→          best = w.axis;
  1146→        }
  1147→      }
  1148→      return best === "h";
  1149→    };
  1150→
  1151→    for (const p of m.points) {
  1152→      const isAirPt = p.kind === "meter_air" || p.kind === "kran" || p.kind === "shower" || p.kind === "kloset" || p.kind === "drain";
  1153→      if (isAirPt && !air) continue;
  1154→      if (!isAirPt && !listrik) continue;
  1155→      if (hid(MEP_POINT_PART[p.kind] ?? p.kind)) continue; // titik bisa disembunyikan per bagian
  1156→      switch (p.kind) {
  1157→        case "lampu":
  1158→          this.addMepSphere(p.x, p.y, p.z, 0.06, this.matMepLampu, "titikLampu");
  1159→          break;
  1160→        case "saklar":
  1161→          if (thinZ(p)) this.addMepBox(p.x, p.y, p.z, 0.09, 0.12, 0.028, this.matMepPlate, "saklar");
  1162→          else this.addMepBox(p.x, p.y, p.z, 0.028, 0.12, 0.09, this.matMepPlate, "saklar");

  1504→function mulberry32(a: number) {
  1505→  return function () {
  1506→    a |= 0;
  1507→    a = (a + 0x6d2b79f5) | 0;
  1508→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
  1509→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
  1510→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  1511→  };
