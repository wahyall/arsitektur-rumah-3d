   544→function clamp01(v: number): number {
   545→  return v < 0 ? 0 : v > 1 ? 1 : v;
   546→}
   547→
   548→interface Interval {
   549→  lo: number;
   550→  hi: number;
   551→}
   552→
   553→function subtractIntervals(base: Interval, cuts: Interval[]): Interval[] {
   554→  let pieces: Interval[] = [{ ...base }];
   555→  for (const c of cuts) {
   556→    const next: Interval[] = [];
   557→    for (const p of pieces) {
   558→      if (c.hi <= p.lo || c.lo >= p.hi) {
   559→        next.push(p);
   560→        continue;
   561→      }
   562→      if (c.lo > p.lo) next.push({ lo: p.lo, hi: c.lo });
   563→      if (c.hi < p.hi) next.push({ lo: c.hi, hi: p.hi });
   564→    }
   565→    pieces = next;
   566→  }
   567→  return pieces.filter((p) => p.hi - p.lo > 0.015); // buang irisan sangat tipis
   568→}
   569→
   570→/**
   571→ * Selesaikan sentinel lubang angin menjadi elevasi nyata.
   572→ * "atas"  -> sill = tinggi dinding - 60 cm (dekat puncak dinding, di bawah ring balk)
   573→ * "pelana"-> di segitiga pelana, menjorok sedikit di bawah puncak apex
   574→ */
   575→function resolveVent(
   576→  o: OpeningSpec,
   577→  t: { rbTop: number; apexAdd: number; tinggi: number },
   578→): OpeningSpec | null {
   579→  if (o.kind !== "vent") return o;
   580→  const hgt = Math.min(0.22, o.to - o.from); // tinggi roster (maks 22 cm)
   581→  if (o.sill === -1) {
   582→    const cap = t.rbTop + t.apexAdd;
   583→    const top = cap - 0.16;
   584→    const sill = top - hgt;
   585→    return sill < t.rbTop + 0.12 ? null : { ...o, sill, top }; // pelana terlalu landai
   586→  }
   587→  const sill = t.tinggi - 0.6;
   588→  return { ...o, sill, top: sill + hgt };
   589→}

   613→  airBersihM: number; // pipa 1/2" + cabang
   614→  airKotorM: number; // pipa 3" ke septic
   615→  conduitM: number; // conduit PVC 20 mm
   616→  kabelLampuM: number; // circuit lampu (NYM 2x1,5)
   617→  kabelKontakM: number; // circuit stop kontak (NYM 2x2,5)
   618→  titikLampu: number;
   619→  saklar: number;
   620→  stopkontak: number;
   621→  kloset: number;
   622→  shower: number;
   623→  kran: number;
   624→  drain: number;
   625→  routes: MepRouteRT[];
   626→  points: MepPointRT[];
   627→  septic: { x: number; z: number };
   628→  resapan: { x: number; z: number };
   629→}
   630→
   631→function polylineLen(pts: { x: number; y: number; z: number }[]): number {
   632→  let L = 0;
   633→  for (let i = 1; i < pts.length; i++) {
   634→    L += Math.hypot(pts[i].x - pts[i - 1].x, pts[i].y - pts[i - 1].y, pts[i].z - pts[i - 1].z);
   635→  }
   636→  return L;
   637→}
   638→
   639→function buildMepRT(spec: MepSpec, tinggi: number): MepRT {
   640→  const routes: MepRouteRT[] = spec.routes.map((r) => {
   641→    const pts = r.pts.map((p) => ({ x: p.x, y: p.rel ? tinggi + p.h : p.h, z: p.z }));
   642→    return { id: r.id, jenis: r.jenis, label: r.label, sirkuit: r.sirkuit ?? null, pts, len: polylineLen(pts) };
   643→  });
   644→  const points: MepPointRT[] = spec.points.map((p) => ({
   645→    id: p.id,
   646→    kind: p.kind,
   647→    label: p.label,
   648→    x: p.x,
   649→    y: p.rel ? tinggi + p.h : p.h,
   650→    z: p.z,
   651→  }));
   652→  const byJenis = (j: MepJenis) => routes.filter((r) => r.jenis === j).reduce((s, r) => s + r.len, 0);
   653→  const cnt = (k: MepKind) => points.filter((p) => p.kind === k).length;
   654→  return {
   655→    airBersihM: byJenis("air_bersih"),
   656→    airKotorM: byJenis("air_kotor"),
   657→    conduitM: byJenis("listrik"),
   658→    kabelLampuM: routes.filter((r) => r.jenis === "listrik" && r.sirkuit !== "kontak").reduce((s, r) => s + r.len, 0),
   659→    kabelKontakM: routes.filter((r) => r.jenis === "listrik" && r.sirkuit === "kontak").reduce((s, r) => s + r.len, 0),
   660→    titikLampu: cnt("lampu"),
   661→    saklar: cnt("saklar"),
   662→    stopkontak: cnt("stopkontak"),
   663→    kloset: cnt("kloset"),
   664→    shower: cnt("shower"),
   665→    kran: cnt("kran"),
   666→    drain: cnt("drain"),
   667→    routes,
   668→    points,
   669→    septic: spec.septic,
   670→    resapan: spec.resapan,
   671→  };
   672→}
   673→
   674→export interface HouseModel {
   675→  config: HouseConfig;
   676→  walls: WallRT[];
   677→  brickDims: { L: number; H: number };
   678→  bricks: BrickInstance[];
   679→  count: {
   680→    outerPieces: number;
   681→    innerPieces: number;
   682→    totalPieces: number;
   683→    buyOuter: number;
   684→    buyInner: number;
   685→    totalBuy: number;
   686→  };
   687→  mortar: {
   688→    volM3: number;
   689→    semenZak: number;
   690→    pasirM3: number;
   691→    perekatKg: number;
   692→    perekatZak: number;
   693→    natMm: number;
   694→  };
   695→  areas: {
   696→    gross: number;
   697→    netFloor: number;
   698→    rooms: RoomNet[];
   699→    reduksiM2: number;
   700→    reduksiPct: number;
   701→    wallFace: number;
   702→    plasterArea: number;
   703→    wallFootprint: number;
   704→  };
   705→  openings: { doors: number; windows: number; vents: number; windowAreaM2: number };
   706→  vent: { bukaanM2: number; pctLantai: number; okSni: boolean; okIdeal: boolean };
   707→  roof: {
   708→    areaM2: number;
   709→    apexAdd: number;
   710→    overhang: number;
   711→    deg: number;
   712→    pitch: number; // tan(deg)
   713→    yRidge: number; // ketinggian nok (m)
   714→    kudaKuda: number;
   715→    rengPerSlope: number;
   716→    gentengPcs: number;
   717→  };
   718→  wallLenTotal: number;
   719→  koloms: KolomRT[];
   720→  mep: MepRT;
   721→  structure: {
   722→    galianM3: number;
   723→    uruganPasirM3: number;
   724→    pondasiM3: number;
   725→    sloofM3: number;
   726→    kolomM3: number;
   727→    ringbalkM3: number;
   728→    lisplankM: number;
   729→    talangM: number;
   730→    cakar: CakarAyamRT | null;
   731→  };
   732→}
   733→
   734→export function buildModel(config: HouseConfig): HouseModel {
   735→  const preset = LAYOUT_PRESETS.find((p) => p.id === config.preset) ?? LAYOUT_PRESETS[0];
   736→  const tL = config.tLuar / 1000;
   737→  const tD = config.tDalam / 1000;
   738→  const dims = BRICK[config.jenis];
   739→  const j = config.natMm / 1000;
   740→
   741→  // geometri atap pelana: ridge memanjang arah Y denah (dinding ujung pelana = depan/belakang)
   742→  const rad = (config.atapDeg * Math.PI) / 180;
   743→  const apexAdd = 3 * Math.tan(rad); // setengah bentang 3 m -> naik ridge
   744→  const pitch = Math.tan(rad);
   745→  const rbTop = config.tinggi + RB_H; // puncak ring balk = dasar dinding pelana
   746→
   747→  // ----- dinding runtime
   748→  const walls: WallRT[] = preset.walls.map((spec) => {
   749→    const t = spec.outer ? tL : tD;
   750→    let c: number;
   751→    if (spec.edge !== undefined) {
   752→      c = spec.edge === 0 ? t / 2 : 6 - t / 2;
   753→    } else {
   754→      c = spec.pos ?? 0;
   755→    }
   756→    const lo = typeof spec.from === "function" ? spec.from({ tL, tD }) : spec.from;
   757→    const hi = typeof spec.to === "function" ? spec.to({ tL, tD }) : spec.to;
   758→    const explodeDir: [number, number] =
   759→      spec.axis === "h"
   760→        ? spec.edge === 6 || (!spec.outer && (spec.pos ?? 0) < 3)
   761→          ? [0, 1]
   762→          : [0, -1]
   763→        : spec.edge === 6 || (!spec.outer && (spec.pos ?? 0) > 3)
   764→          ? [1, 0]
   765→          : [-1, 0];
   766→    return {
   767→      spec,
   768→      id: spec.id,
   769→      name: spec.name,
   770→      axis: spec.axis,
   771→      c,
   772→      lo,
   773→      hi,
   774→      t,
   775→      h: config.tinggi,
   776→      gable: spec.gable ? { ...spec.gable, apexAdd, baseZ: rbTop } : undefined,
   777→      openings: spec.openings
   778→        .map((o) => resolveVent(o, { rbTop, apexAdd, tinggi: config.tinggi }))
   779→        .filter((o): o is OpeningSpec => !!o)
   780→        .filter((o) => o.to > lo && o.from < hi && o.top > 0.2),
   781→      outer: spec.outer,
   782→      explodeDir,
   783→    };
   784→  });
   785→
   786→  // ----- susunan bata per dinding
   787→  const bricks: BrickInstance[] = [];
   788→  let outerPieces = 0;
   789→  let innerPieces = 0;
   790→  let wallFace = 0;
   791→  let bandVol = 0;
   792→  let brickVol = 0;
   793→  let wallFootprint = 0;
   794→  let wallLenTotal = 0;
   795→
   796→  const cellL = dims.L + j;
   797→  const cellH = dims.H + j;
   798→
   799→  for (const w of walls) {
   800→    const len = w.hi - w.lo;
   801→    wallLenTotal += len;
   802→    wallFootprint += len * w.t;
   803→
   804→    const hAt = w.gable
   805→      ? (x: number) => w.gable!.baseZ + w.gable!.apexAdd * clamp01(1 - Math.abs(x - w.gable!.apexAt) / w.gable!.halfSpan)
   806→      : () => w.h;
   807→
   808→    // luas muka satu sisi (net bukaan: tinggi bukaan = top - sill, dipotong tinggi
   809→    // dinding datar / pelana; band ring balk tidak dihitung karena beton)
   810→    const gableArea = w.gable ? w.gable.apexAdd * (w.hi - w.lo) * 0.5 : 0;
   811→    const hAtMid = (o: OpeningSpec) =>
   812→      Math.max(w.h, hAt((o.from + o.to) / 2));
   813→    const openingArea = w.openings.reduce(
   814→      (s, o) =>
   815→        s +
   816→        Math.max(0, Math.min(o.to, w.hi) - Math.max(o.from, w.lo)) *
   817→          Math.max(0, Math.min(o.top, hAtMid(o)) - o.sill),
   818→      0,
   819→    );
   820→    wallFace += w.h * len + gableArea - openingArea;
   821→    bandVol += (w.h * len + gableArea) * w.t - openingArea * w.t;
   822→
   823→    const topZ = w.gable ? w.gable.baseZ + w.gable.apexAdd : w.h;
   824→    const nCourses = Math.ceil(topZ / cellH);
   825→    for (let i = 0; i < nCourses; i++) {
   826→      const zc = i * cellH;
   827→      if (zc >= topZ - 0.001) break;
   828→
   829→      // interval yang boleh dipasangi bata pada lajur ini:
   830→      // 1) band ring balk (w.h .. baseZ) tidak dipasangi bata; di atas baseZ
   831→      //    bata mengikuti penyempitan segitiga pelana
   832→      if (w.gable && zc >= w.h && zc < w.gable.baseZ) continue; // zona ring balk
   833→
   834→      const allowed: Interval = { lo: w.lo, hi: w.hi };
   835→      if (w.gable && zc >= w.gable.baseZ) {
   836→        const d = w.gable.halfSpan * (1 - (zc - w.gable.baseZ) / w.gable.apexAdd);
   837→        if (d <= 0) continue;
   838→        allowed.lo = Math.max(w.lo, w.gable.apexAt - d);
   839→        allowed.hi = Math.min(w.hi, w.gable.apexAt + d);
   840→      }
   841→      if (allowed.hi - allowed.lo <= 0.015) continue;
   842→
   843→      // 2) bukaan yang memotong lajur ini secara vertikal
   844→      const activeCuts: Interval[] = w.openings
   845→        .filter((o) => o.sill < zc + dims.H - 0.005 && o.top > zc + 0.005)
   846→        .map((o) => ({ lo: Math.max(o.from, w.lo), hi: Math.min(o.to, w.hi) }));
   847→
   848→      // 3) isi bata dengan ikatan berselang (stretcher bond)
   849→      const phase = (i % 2) * cellL * 0.5;
   850→      let p = allowed.lo - phase;
   851→      while (p + dims.L <= allowed.lo) p += cellL;
   852→      for (; p < allowed.hi; p += cellL) {
   853→        const segs = subtractIntervals(
   854→          { lo: Math.max(p, allowed.lo), hi: Math.min(p + dims.L, allowed.hi) },
   855→          activeCuts,
   856→        );
   857→        for (const s of segs) {
   858→          const cx = (s.lo + s.hi) / 2;
   859→          // di bawah ring balk dipotong rata di tinggi dinding; di atas mengikuti kemiringan pelana
   860→          const hgt =
   861→            zc < w.h
   862→              ? Math.min(dims.H, Math.max(0, w.h - zc))
   863→              : Math.min(dims.H, Math.max(0, hAt(cx) - zc));
   864→          if (hgt <= 0.02) continue;
   865→          const frac = ((s.hi - s.lo) / dims.L) * (hgt / dims.H);
   866→          if (w.outer) outerPieces += frac;
   867→          else innerPieces += frac;
   868→          brickVol += frac * dims.L * dims.H * w.t;
   869→          bricks.push(
   870→            w.axis === "h"
   871→              ? { wallId: w.id, outer: w.outer, x: cx, y: w.c, z: zc, len: s.hi - s.lo, hgt, rotY: 0 }
   872→              : { wallId: w.id, outer: w.outer, x: w.c, y: cx, z: zc, len: s.hi - s.lo, hgt, rotY: Math.PI / 2 },
   873→          );
   874→        }
   875→      }
   876→    }
   877→  }
   878→
   879→  const mortarVol = Math.max(0, bandVol - brickVol);
   880→
   881→  // ----- luas bersih ruangan (setelah dikurangi tebal dinding)
   882→  interface Band {
   883→    axis: Axis;
   884→    x0: number;
   885→    x1: number;
   886→    y0: number;
   887→    y1: number;
   888→  }
   889→  const bands: Band[] = walls.map((w) =>
   890→    w.axis === "h"
   891→      ? { axis: w.axis, x0: w.lo, x1: w.hi, y0: w.c - w.t / 2, y1: w.c + w.t / 2 }
   892→      : { axis: w.axis, x0: w.c - w.t / 2, x1: w.c + w.t / 2, y0: w.lo, y1: w.hi },
   893→  );
   894→  const rooms: RoomNet[] = preset.rooms.map((r) => {
   895→    let xl = 0, xr = 0, yb = 0, yt = 0;
   896→    for (const b of bands) {
   897→      const yOverlap = Math.min(b.y1, r.y1) - Math.max(b.y0, r.y0);
   898→      const xOverlap = Math.min(b.x1, r.x1) - Math.max(b.x0, r.x0);
   899→      if (b.axis === "v") {
   900→        if (b.x0 <= r.x0 + 0.02 && b.x1 > r.x0 && yOverlap > 0.05) xl = Math.max(xl, b.x1 - r.x0);
   901→        if (b.x1 >= r.x1 - 0.02 && b.x0 < r.x1 && yOverlap > 0.05) xr = Math.max(xr, r.x1 - b.x0);
   902→      } else {
   903→        if (b.y0 <= r.y0 + 0.02 && b.y1 > r.y0 && xOverlap > 0.05) yb = Math.max(yb, b.y1 - r.y0);
   904→        if (b.y1 >= r.y1 - 0.02 && b.y0 < r.y1 && xOverlap > 0.05) yt = Math.max(yt, r.y1 - b.y0);
   905→      }
   906→    }
   907→    const wNet = Math.max(0, r.x1 - r.x0 - xl - xr);
   908→    const dNet = Math.max(0, r.y1 - r.y0 - yb - yt);
   909→    return { id: r.id, name: r.name, w: wNet, d: dNet, area: wNet * dNet, cx: (r.x0 + r.x1) / 2, cy: (r.y0 + r.y1) / 2 };
   910→  });
   911→
   912→  const netFloor = rooms.reduce((s, r) => s + r.area, 0);
   913→  const gross = 36;
   914→
   915→  // ----- kolom praktis 15/15: pada tiap pertemuan centerline dinding h x v
   916→  const seen = new Set<string>();
   917→  const koloms: KolomRT[] = [];
   918→  const tol = 0.15;
   919→  for (const h of walls) {
   920→    if (h.axis !== "h") continue;
   921→    for (const v of walls) {
   922→      if (v.axis !== "v") continue;
   923→      if (v.c >= h.lo - tol && v.c <= h.hi + tol && h.c >= v.lo - tol && h.c <= v.hi + tol) {
   924→        const key = `${v.c.toFixed(2)}|${h.c.toFixed(2)}`;
   925→        if (!seen.has(key)) {
   926→          seen.add(key);
   927→          koloms.push({ x: v.c, y: h.c });
   928→        }
   929→      }
   930→    }
   931→  }
   932→
   933→  // ----- material
   934→  const waste = config.wastePct / 100;
   935→  const mortarWithWaste = mortarVol * (1 + waste);
   936→  const semenZak = config.jenis === "merah" ? mortarWithWaste * 6.3 : 0;
   937→  const pasirM3 = config.jenis === "merah" ? mortarWithWaste * 1.0 : 0;
   938→  const perekatKg = config.jenis === "hebel" ? wallFace * 4 : 0;
   939→  const perekatZak = config.jenis === "hebel" ? perekatKg / 40 : 0;
   940→
   941→  // ----- atap pelana (kuda-kuda C75 jarak maks 1,2 m, genteng beton 10 pcs/m2)
   942→  const ov = config.overhang;
   943→  const yRidge = rbTop + apexAdd;
   944→  const slopeHalf = Math.hypot(3 + ov, apexAdd + ov * pitch);
   945→  const roofArea = 2 * slopeHalf * (6 + 2 * ov);
   946→  // jumlah kuda-kuda: seluruh bentang 6 m dengan jarak antar rangka maks 1,2 m
   947→  const kudaKuda = Math.max(4, Math.ceil(6 / KUDA_SPACING) + 1);
   948→  const rengPerSlope = Math.ceil(3 / 0.4) + 1;
   949→  const gentengPcs = Math.ceil(roofArea * 10);
   950→
   951→  const doors = walls.reduce((s, w) => s + w.openings.filter((o) => o.kind === "pintu").length, 0);
   952→  const windows = walls.reduce((s, w) => s + w.openings.filter((o) => o.kind === "jendela").length, 0);
   953→  const vents = walls.reduce((s, w) => s + w.openings.filter((o) => o.kind === "vent").length, 0);
   954→  const windowAreaM2 = walls.reduce(
   955→    (s, w) => s + w.openings.filter((o) => o.kind === "jendela").reduce((a, o) => a + (o.to - o.from) * (o.top - o.sill), 0),
   956→    0,
   957→  );
   958→
   959→  // ----- pondasi cakar ayam: buis di tiap titik kolom + grid tengah (jarak maks ~3 m)
   960→  let cakar: CakarAyamRT | null = null;
   961→  if (config.pondasi === "cakarayam") {
   962→    const pts: { x: number; y: number }[] = koloms.map((k) => ({ x: k.x, y: k.y }));
   963→    for (const gx of [1.5, 4.5]) for (const gy of [1.5, 4.5]) pts.push({ x: gx, y: gy });
   964→    const buisPts: { x: number; y: number }[] = [];
   965→    for (const p of pts) {
   966→      if (!buisPts.some((q) => Math.hypot(q.x - p.x, q.y - p.y) < 0.45)) buisPts.push(p);
   967→    }
   968→    const rOut = CAKAR.buisDia / 2;
   969→    const rIn = rOut - CAKAR.buisWall;
   970→    const buisM3 = buisPts.length * Math.PI * (rOut * rOut - rIn * rIn) * CAKAR.buisL;
   971→    const slabM3 = 36 * CAKAR.slabT;
   972→    cakar = {
   973→      slabM3,
   974→      buisCount: buisPts.length,
   975→      buisM3,
   976→      uruganM3: 36 * CAKAR.bed,
   977→      galianM3: 36 * CAKAR.bed + buisPts.length * Math.PI * rOut * rOut * CAKAR.buisL,
   978→      betonM3: slabM3 + buisM3,
   979→      buisPts,
   980→    };
   981→  }
   982→
   983→  // ----- kuantitas struktur (galian & pondasi mengikuti jenis pondasi)
   984→  const galianM3 =
   985→    config.pondasi === "cakarayam" && cakar
   986→      ? cakar.galianM3
   987→      : wallLenTotal * PONDASI_D * GALIAN_W; // galian 60 x 60 cm
   988→  const uruganPasirM3 = config.pondasi === "cakarayam" ? 0 : wallLenTotal * GALIAN_W * URUGAN_T;
   989→  const pondasiM3 = config.pondasi === "cakarayam" ? 0 : wallLenTotal * PONDASI_W * PONDASI_D;
   990→  const sloofM3 = wallLenTotal * SLOOF_W * RB_H;
   991→  const kolomM3 = koloms.length * KOLOM_S * KOLOM_S * (config.tinggi + RB_H);
   992→  const ringbalkM3 = wallLenTotal * SLOOF_W * RB_H;
   993→  const lisplankM = 2 * (6 + 2 * ov) + 2 * slopeHalf; // eaves + kedua ujung pelana
   994→  const talangM = 2 * (6 + 2 * ov);
   995→
   996→  return {
   997→    config,
   998→    walls,
   999→    brickDims: { L: dims.L, H: dims.H },
  1000→    bricks,
  1001→    count: {
  1002→      outerPieces,
  1003→      innerPieces,
  1004→      totalPieces: outerPieces + innerPieces,
  1005→      buyOuter: Math.ceil(outerPieces * (1 + waste)),
  1006→      buyInner: Math.ceil(innerPieces * (1 + waste)),
  1007→      totalBuy: Math.ceil(outerPieces * (1 + waste)) + Math.ceil(innerPieces * (1 + waste)),
  1008→    },
  1009→    mortar: {
  1010→      volM3: mortarVol,
  1011→      semenZak,
  1012→      pasirM3,
  1013→      perekatKg,
  1014→      perekatZak,
  1015→      natMm: config.natMm,
  1016→    },
  1017→    areas: {
  1018→      gross,
  1019→      netFloor,
  1020→      rooms,
  1021→      reduksiM2: gross - netFloor,
  1022→      reduksiPct: ((gross - netFloor) / gross) * 100,
  1023→      wallFace,
  1024→      plasterArea: wallFace * 2,
  1025→      wallFootprint,
  1026→    },
  1027→    openings: { doors, windows, vents, windowAreaM2 },
  1028→    vent: {
  1029→      bukaanM2: windowAreaM2,
  1030→      pctLantai: (windowAreaM2 / Math.max(0.01, netFloor)) * 100,
  1031→      okSni: windowAreaM2 >= 0.05 * netFloor,
  1032→      okIdeal: windowAreaM2 >= 0.1 * netFloor,
  1033→    },
  1034→    roof: { areaM2: roofArea, apexAdd, overhang: ov, deg: config.atapDeg, pitch, yRidge, kudaKuda, rengPerSlope, gentengPcs },

  1146→export function buildRab(model: HouseModel, prices: PriceBook, upahPct: number): RabSummary {
  1147→  const cfg = model.config;
  1148→  const rows: RabRow[] = [];
  1149→  const push = (id: string, divisi: string, uraian: string, spek: string, qty: number, unit: string, exact: boolean, priceKey?: string) => {

  1152→    rows.push({ id, divisi, uraian, spek, qty, unit, harga, subtotal: qty * harga, exact, priceKey: key });
